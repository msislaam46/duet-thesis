This appendix collects the eight algorithms the thesis refers to. They fall into two groups.
Algorithms~\ref{alg:merklelog}--\ref{alg:dualindex} are the data-structure and derivation
primitives: the Merkle log tree, the sparse prefix tree, the key-transparency server that combines
them, and the dual-index label derivation. Algorithms~\ref{alg:client}--\ref{alg:cea} are the
protocol procedures built on those primitives: client verification, MitM detection and warning
persistence, the auditor, and the continuous-authentication mode. \Cref{tab:alg-purpose} states the
purpose of each.

\paragraph{Notation.} $\Hash(\cdot)$ is SHA-256. Tree hashing is domain-separated per RFC 6962/9162: a
leaf hashes as $\Hleaf(x)=\Hash(\texttt{0x00}\concat x)$ and an internal node as
$\Hnode(l,r)=\Hash(\texttt{0x01}\concat l\concat r)$. The operator $\concat$ denotes byte
concatenation. Identities are fixed-width 16-byte service IDs; this is a precondition of the
dual-index derivation (Algorithm~\ref{alg:dualindex}), since without a fixed width the
$\mathit{id_{self}}\concat\mathit{id_{peer}}$ split would be ambiguous, and the reference
implementation enforces it at the API boundary.

\begin{algorithm}[H]
\caption{Merkle Log Tree (RFC 9162)}\label{alg:merklelog}
\KwData{$\mathit{leaves}[]\gets[\,]$}
\Fn{\normalfont\textsf{Append}($\mathit{record}$)}{
  $\mathit{leaves}.\textsf{append}(\Hleaf(\mathit{record}))$\;
  \KwRet $|\mathit{leaves}|-1$ \tcp*{index of new leaf}
}
\Fn{\normalfont\textsf{MTH}($D[a{:}b]$)}{
  $n\gets b-a$\;
  \lIf{$n=0$}{\KwRet $\Hash(\varepsilon)$}
  \lIf{$n=1$}{\KwRet $\mathit{leaves}[a]$}
  $k\gets$ largest power of two $<n$ \tcp*{RFC split point}
  \KwRet $\Hnode(\textsf{MTH}(D[a{:}a{+}k]),\textsf{MTH}(D[a{+}k{:}b]))$\;
}
\Fn{\normalfont\textsf{Root}($\mathit{tree\_size}$)}{
  \KwRet $\textsf{MTH}(D[0{:}\mathit{tree\_size}])$\;
}
\Fn{\normalfont\textsf{InclusionProof}($m,\mathit{tree\_size}$)}{
  \Req $0\le m<\mathit{tree\_size}$\;
  $\mathit{path}\gets$ sibling subtree hashes along RFC splits, leaf-to-root\;
  \KwRet $\mathit{path}$\;
}
\Fn{\normalfont\textsf{VerifyInclusion}($\mathit{record},m,\mathit{tree\_size},\mathit{root},\mathit{path}$)}{
  \Req $0\le m<\mathit{tree\_size}$\;
  $\mathit{fn}\gets m$;\ \ $\mathit{sn}\gets \mathit{tree\_size}-1$;\ \ $r\gets \Hleaf(\mathit{record})$\;
  \ForEach{$p\in\mathit{path}$}{
    \Req $\mathit{sn}\ne 0$\;
    \uIf{$\mathit{fn}$ odd $\vee\ \mathit{fn}=\mathit{sn}$}{
      $r\gets \Hnode(p,r)$\;
      \lIf{$\mathit{fn}$ even}{right-shift $\mathit{fn},\mathit{sn}$ until $\mathit{fn}$ odd $\vee\ \mathit{fn}=0$}
    }
    \lElse{$r\gets \Hnode(r,p)$}
    $\mathit{fn}\gets\lfloor \mathit{fn}/2\rfloor$;\ \ $\mathit{sn}\gets\lfloor \mathit{sn}/2\rfloor$\;
  }
  \KwRet $\mathit{sn}=0\wedge r=\mathit{root}$ \tcp*{rejects short/padded paths}
}
\Fn{\normalfont\textsf{ConsistencyProof}($m,n$)}{
  \Req $0\le m\le n$\;
  \lIf{$m=0\vee m=n$}{\KwRet $[\,]$}
  \KwRet $\textsf{SUBPROOF}(m,D[0{:}n],\mathrm{true})$ \tcp*{\S2.1.4.1}
}
\Fn{\normalfont\textsf{VerifyConsistency}($m,n,\mathit{old\_root},\mathit{new\_root},\mathit{proof}$)}{
  \lIf{$m=n$}{\KwRet $\mathit{old\_root}=\mathit{new\_root}\wedge \mathit{proof}=[\,]$}
  \lIf{$m=0$}{\KwRet $\mathit{proof}=[\,]$}
  \Req $\mathit{proof}\ne[\,]$\;
  \lIf{$m$ is a power of two}{$\mathit{proof}\gets \mathit{old\_root}:\mathit{proof}$}
  $\mathit{fn}\gets m-1$;\ \ $\mathit{sn}\gets n-1$\;
  right-shift $\mathit{fn},\mathit{sn}$ while $\mathit{fn}$ odd\;
  $\mathit{fr}\gets\mathit{proof}[0]$;\ \ $\mathit{sr}\gets\mathit{proof}[0]$\;
  fold remaining proof entries into $(\mathit{fr},\mathit{sr})$ per \S2.1.4.2\;
  \KwRet $\mathit{fr}=\mathit{old\_root}\wedge \mathit{sr}=\mathit{new\_root}\wedge \mathit{sn}=0$\;
}
\end{algorithm}

\begin{algorithm}[H]
\caption{Sparse Binary Merkle Prefix Tree (label-keyed)}\label{alg:prefix}
\KwData{path-compressed binary trie over 256-bit labels; empty hash $e=\Hash(\varepsilon)$}
\Fn{\normalfont\textsf{Insert}($\mathit{label},\mathit{value},\mathit{kind}$)}{
  \Req $|\mathit{label}|=32$ \tcp*{$\mathit{kind}\in\{\textsc{mutable},\textsc{immutable}\}$, bound into the leaf (C2)}
  place leaf $\Hleaf(\mathit{kind}\concat\mathit{label}\concat\mathit{value})$ at the bit-path of $\mathit{label}$\;
  recompute hashes along the insertion path only \tcp*{$O(256)$}
  \KwRet \textsf{RootHash}()\;
}
\Fn{\normalfont\textsf{Lookup}($\mathit{label}$)}{
  walk the bit-path of $\mathit{label}$\;
  $\pi\gets(\mathit{label},\mathit{value}\text{ or }\bot,\mathit{kind},\mathit{bitmap},\mathit{siblings})$ \tcp*{compressed proof}
  \KwRet $(\mathit{value}\text{ or }\bot,\ \pi)$\;
}
\Fn{\normalfont\textsf{RootFromProof}($\pi$)}{
  $c\gets e$ \textbf{if} $\pi.\mathit{value}=\bot$ \textbf{else} $\Hleaf(\pi.\mathit{kind}\concat\pi.\mathit{label}\concat\pi.\mathit{value})$ \tcp*{kind authenticated by the root (C2)}\;
  \For{$i\gets 0$ \KwTo $255$}{
    $s\gets$ next sibling from $\pi.\mathit{siblings}$ if $\mathit{bitmap}[i]$ set, else $e$\;
    \lIf{$c=e\wedge s=e$}{\textbf{continue}}
    $c\gets \Hnode(c,s)$ or $\Hnode(s,c)$ per bit $255-i$ of $\pi.\mathit{label}$\;
  }
  \Req all siblings consumed, else \KwRet $\bot$\;
  \KwRet $c$\;
}
\Fn{\normalfont\textsf{VerifyProof}($\pi,\mathit{expected\_root}$)}{
  \KwRet $\textsf{RootFromProof}(\pi)=\mathit{expected\_root}$\;
}
\end{algorithm}

\begin{algorithm}[H]
\caption{Key Transparency Server}\label{alg:server}
\KwData{$\mathit{log}\gets$ MerkleLogTree, $\mathit{live}\gets$ PrefixTree, $\mathit{committed}\gets\bot$, $\mathit{epoch}\gets0$, $(sk_{\mathrm{vrf}},pk_{\mathrm{vrf}})$, $(sk_{\mathrm{sth}},pk_{\mathrm{sth}})$}
\Fn{\normalfont\textsf{UpdateByUserId}($\mathit{user\_id},pk$)}{
  $(\beta,\pi_{\mathrm{vrf}})\gets \mathsf{VRF.Prove}(sk_{\mathrm{vrf}},\mathit{user\_id})$ \tcp*{ECVRF, RFC 9381}
  $\mathit{label}\gets\beta[0{:}32]$ \tcp*{32-byte truncation}
  $\mathit{live}.\textsf{Insert}(\mathit{label},pk,\textsc{mutable})$ \tcp*{global-VRF identity: leaf hash $\Hleaf(\textsc{mutable}\concat\mathit{label}\concat pk)$}
  record \textsf{Insert}/\textsf{Rotate} with $\mathit{kind}=\textsc{mutable}$ in the epoch's declared delta\;
  \KwRet $(\mathit{label},\pi_{\mathrm{vrf}})$\;
}
\Fn{\normalfont\textsf{UpdateRaw}($\mathit{label},\mathit{value}$)}{
  \Req $\mathit{live}.\textsf{Lookup}(\mathit{label}).\mathit{value}=\bot$ \tcp*{write-once: occupied labels rejected (\cref{ssec:design-detection})}
  $\mathit{live}.\textsf{Insert}(\mathit{label},\mathit{value},\textsc{immutable})$ \tcp*{dual-index/CEA leaf $\Hleaf(\textsc{immutable}\concat\mathit{label}\concat\mathit{value})$}
  record \textsf{Insert}$(\mathit{label},\mathit{value},\textsc{immutable})$ in the epoch's declared delta\;
}
\Fn{\normalfont\textsf{CommitEpoch}()}{
  $r\gets \mathit{live}.\textsf{RootHash}()$\;
  $i\gets \mathit{log}.\textsf{Append}(r)$ \tcp*{prefix root is a log leaf}
  $\mathit{committed}\gets(\textsf{snapshot}(\mathit{live}),i)$\;
  freeze the declared delta for this epoch \tcp*{served to auditors via \textsf{GetEpochDelta} (Alg.~\ref{alg:auditor})}
  $\mathit{epoch}\gets \mathit{epoch}+1$\;
  $\mathit{ch}\gets \Hash(pk_{\mathrm{vrf}}\concat pk_{\mathrm{sth}}\concat \mathit{suite}\concat \mathit{version}\concat \mathit{interval}\concat \mathit{rawpolicy})$ \tcp*{policy commitment (CONIKS $P$); $\mathit{rawpolicy}$ binds the write-once/delta policy (H3)}
  $\sigma\gets \textsf{Sign}(sk_{\mathrm{sth}},\ \mathit{epoch}{-}1\concat \mathit{log}.\mathit{size}\concat \mathit{log}.\textsf{Root}\concat \mathit{ts}\concat \mathit{ch})$ \tcp*{88-byte LE}
  $\mathit{sth}\gets(\mathit{epoch}{-}1,\mathit{log}.\mathit{size},\mathit{log}.\textsf{Root},\mathit{ts},\mathit{ch},\sigma)$\;
  \KwRet $\mathit{sth}$\;
}
\Fn{\normalfont\textsf{Search}($\mathit{label}$)}{
  \Req $\mathit{committed}\ne\bot$\;
  $(v,\pi_{\mathrm{prefix}})\gets \mathit{committed}.\mathit{tree}.\textsf{Lookup}(\mathit{label})$\;
  $\pi_{\mathrm{incl}}\gets \mathit{log}.\textsf{InclusionProof}(\mathit{committed}.i,\mathit{log}.\mathit{size})$\;
  \KwRet $(v,\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}},\mathit{sth}_{\mathrm{latest}})$\;
}
\Fn{\normalfont\textsf{SearchByUserId}($\mathit{user\_id}$)}{
  $(\beta,\pi_{\mathrm{vrf}})\gets \mathsf{VRF.Prove}(sk_{\mathrm{vrf}},\mathit{user\_id})$\;
  \KwRet $\textsf{Search}(\beta[0{:}32])$ with $\pi_{\mathrm{vrf}}$ attached\;
}
\end{algorithm}

\begin{algorithm}[H]
\caption{Dual-Index Label Derivation (HKDF-SHA256, RFC 5869)}\label{alg:dualindex}
\KwIn{shared secret $\sharedS$ from X3DH; identities $\mathit{id_{self}},\mathit{id_{peer}}$ (fixed-width); $\min/\max$ below denote the canonically ordered conversation pair $(\mathit{id_A},\mathit{id_B})$}
\Fn{\normalfont\textsf{DeriveLabels}($\sharedS,\mathit{id_{self}},\mathit{id_{peer}}$)}{
  $\mathit{PRK}\gets \textsf{HKDF-Extract}(\text{salt}=\texttt{"duet-dual-index"},\ \sharedS)$\;
  $\lself\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"lookup"}\concat \mathit{id_{self}}\concat \mathit{id_{peer}},32)$\;
  $\lpeer\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"lookup"}\concat \mathit{id_{peer}}\concat \mathit{id_{self}},32)$\;
  $(\mathit{lo},\mathit{hi})\gets \textsf{sort}(\mathit{id_{self}},\mathit{id_{peer}})$ \tcp*{canonical order: warning only}
  $\lwarn\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"warning"}\concat \mathit{lo}\concat \mathit{hi},32)$\;
  \KwRet $(\lself,\lpeer,\lwarn)$\;
}
\Fn{\normalfont\textsf{WriteWarning}($\mathit{server},\lwarn,\mathit{alert},\mathit{PRK}$)}{
  \lIf{$\mathit{server}$ already holds a value at $\lwarn$}{\KwRet \tcp*[f]{slot is write-once; an existing value already alerts both monitors}}
  $k_{\mathrm{mac}}\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"warn-mac"}\concat\min(\mathit{id_A},\mathit{id_B})\concat\max(\mathit{id_A},\mathit{id_B}),32)$ \tcp*{canonical identities bind $k_{\mathrm{mac}}$ to the pair; only the two parties can compute it}
  $\mathit{server}.\textsf{UpdateRaw}(\lwarn,\ \mathit{alert}\concat\textsf{MAC}(k_{\mathrm{mac}},\mathit{alert}))$\;
  after the next epoch commits: $\textsf{RecordWarning}(\lwarn)$ \tcp*{retain evidence (Alg.~\ref{alg:detect})}
}
\Fn{\normalfont\textsf{CheckWarning}($\mathit{server},\lwarn,\mathit{sth},\mathit{PRK}$)}{
  $v\gets \textsf{VerifiedLookup}(\mathit{server},\lwarn,\mathit{sth})$ \tcp*{proof required (Alg.~\ref{alg:client})}
  \lIf{$v=\bot$}{\KwRet \textsc{Safe}}
  $k_{\mathrm{mac}}\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"warn-mac"}\concat\min(\mathit{id_A},\mathit{id_B})\concat\max(\mathit{id_A},\mathit{id_B}),32)$\;
  \lIf{$\textsf{MAC-Verify}(k_{\mathrm{mac}},v)$}{\KwRet \textsc{Alert} \tcp*[f]{authentic peer warning}}
  \KwRet \textsc{Alert} \tcp*{unauthenticated data at an unpredictable slot: itself evidence of misbehaviour}
}
\end{algorithm}

\paragraph{Client verification (Algorithm~\ref{alg:client}).} These procedures are the client's entire
trust boundary: every value it accepts is checked against a signed tree head before use.
\textsf{VerifiedLookup} returns a value only when the prefix proof, the log-inclusion proof, and the
STH signature all verify and bind to the caller's pinned head, and it fails closed on any mismatch.
\textsf{Register}, \textsf{RetrieveKey}, and \textsf{WindowedCatchUp} build on it to establish, read,
and re-synchronize client state.

\begin{algorithm}[H]
\caption{Client Verification, Registration, Retrieval, Windowed Catch-Up}\label{alg:client}
\KwData{$\mathit{last\_sth}\gets\bot$; per conversation, $\mathit{PRK}$ as extracted in Alg.~\ref{alg:dualindex}}
\Fn{\normalfont\textsf{VerifiedLookup}($\mathit{server},\mathit{label},\mathit{sth},[\mathit{user\_id},\pi_{\mathrm{vrf}}]$)}{
  $(v,\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}},\mathit{sth'})\gets \mathit{server}.\textsf{Search}(\mathit{label})$\;
  \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth'})$ \tcp*{Ed25519 over epoch$\concat$size$\concat$root$\concat$ts$\concat$H(config)}
  \Req $\mathit{sth'}=\mathit{sth}$ \tcp*{bind to the caller's verified head; a signed but stale/forked head is rejected}
  \Req $\pi_{\mathrm{prefix}}.\mathit{label}=\mathit{label}$ \tcp*{label binding (every path): the proof must speak about the requested label, so a valid proof for a different label is rejected}
  \If{$\mathit{user\_id}$ given}{
    $\beta\gets \mathsf{VRF.Verify}(pk_{\mathrm{vrf}},\mathit{user\_id},\pi_{\mathrm{vrf}})$;\ \ \Req $\beta[0{:}32]=\mathit{label}$ \tcp*[f]{VRF path: the derived label equals the requested one}\;
  }
  \Req $\pi_{\mathrm{prefix}}.\mathit{value}=v$\;
  $r\gets \textsf{RootFromProof}(\pi_{\mathrm{prefix}})$;\ \ \Req $r\ne\bot$\;
  \Req $\textsf{VerifyInclusion}(r,\pi_{\mathrm{incl}}.\mathit{index},\pi_{\mathrm{incl}}.\mathit{tree\_size},\mathit{sth'}.\mathit{root},\pi_{\mathrm{incl}}.\mathit{path})$\;
  \Req $\pi_{\mathrm{incl}}.\mathit{tree\_size}=\mathit{sth'}.\mathit{tree\_size}\ \wedge\ \pi_{\mathrm{incl}}.\mathit{index}=\mathit{sth'}.\mathit{tree\_size}-1$ \tcp*{currency: the proof must cover the last log leaf; an older leaf's inclusion also verifies (inclusion $\ne$ currency, RFC 9162)}
  \KwRet $v$ \tcp*{$v$ may be $\bot$: proven non-inclusion}
}
\Fn{\normalfont\textsf{Register}($\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}},pk_{\mathit{self}}$)}{
  $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
  $\mathit{server}.\textsf{UpdateRaw}(\lself,pk_{\mathit{self}})$\;
  $\mathit{sth}\gets \mathit{server}.\textsf{CommitEpoch}()$ \tcp*{PoC only; a deployed client waits for the epoch boundary}
  $v\gets \textsf{VerifiedLookup}(\mathit{server},\lself,\mathit{sth})$;\ \ \Req $v=pk_{\mathit{self}}$\;
  $\mathit{last\_sth}\gets \mathit{sth}$\;
  \KwRet $(\lself,\lpeer,\lwarn)$\;
}
\Fn{\normalfont\textsf{RetrieveKey}($\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}}$)}{
  $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
  $\textsf{WindowedCatchUp}(\mathit{server},\lself,\lwarn,\mathit{PRK})$ \tcp*{advance $\mathit{last\_sth}$ to the current verified head: every lookup binds to it}
  $pk\gets \textsf{VerifiedLookup}(\mathit{server},\lpeer,\mathit{last\_sth})$;\ \ \Req $pk\ne\bot$\;
  \KwRet $pk$\;
}
\Fn{\normalfont\textsf{WindowedCatchUp}($\mathit{server},\lself,\lwarn,\mathit{PRK}$)}{
  $\mathit{sth_{new}}\gets \mathit{server}.\textsf{GetCurrentSTH}()$;\ \ \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth_{new}})$\;
  \uIf{$\mathit{last\_sth}=\bot$}{
    $\mathit{last\_sth}\gets \mathit{sth_{new}}$ \tcp*{first run: trust-on-first-use anchor (\cref{sec:threat-fair}); no history to check}
  }\Else{
    \lIf{$\mathit{sth_{new}}.\mathit{epoch}<\mathit{last\_sth}.\mathit{epoch} \lor \mathit{sth_{new}}.\mathit{size}<\mathit{last\_sth}.\mathit{size}$}{\textbf{raise} Fork}
    \tcc{an honest append-only head never regresses in epoch or size}
    \uIf{$\mathit{sth_{new}}.\mathit{size}=\mathit{last\_sth}.\mathit{size}$}{
      \Req $\mathit{sth_{new}}.\mathit{root}=\mathit{last\_sth}.\mathit{root}$ \tcp*{unchanged head: equal size with a differing root is same-epoch equivocation, else \textbf{raise} Fork; only the consistency proof is skipped}
    }\Else{
      $\pi_{\mathrm{cons}}\gets \mathit{server}.\textsf{ConsistencyProof}(\mathit{last\_sth}.\mathit{size},\mathit{sth_{new}}.\mathit{size})$\;
      \Req $\textsf{VerifyConsistency}(\mathit{last\_sth}.\mathit{size},\mathit{sth_{new}}.\mathit{size},\mathit{last\_sth}.\mathit{root},\mathit{sth_{new}}.\mathit{root},\pi_{\mathrm{cons}})$\;
      \tcc{one proof spans the whole window: $O(\log n)$ rather than $O(W\log n)$}
    }
  }
  \tcc{the self-key lookup and warning check below run on every tick, including an unchanged head}
  $v\gets \textsf{VerifiedLookup}(\mathit{server},\lself,\mathit{sth_{new}})$;\ \ \Req $v\ne\bot$\;
  \lIf{$\textsf{CheckWarning}(\mathit{server},\lwarn,\mathit{sth_{new}},\mathit{PRK})=\textsc{Alert}$}{\textbf{raise} SecurityWarning}
  $\mathit{last\_sth}\gets \mathit{sth_{new}}$\;
}
\end{algorithm}

\paragraph{Detection and persistence (Algorithm~\ref{alg:detect}).} This is the monitoring loop that
turns a substituted key into an alert. \textsf{DetectMitM} and \textsf{BackgroundMonitor} compare the
retrieved key against the expected one and, on a mismatch, write the dual-index warning.
\textsf{RecordWarning} and \textsf{PersistenceCheck} retain verified evidence, so that a later silent
deletion of a committed warning becomes publishable proof rather than an unobservable loss. No path
silences a genuine mismatch: the outcome is always a raised alert or retained evidence.

\begin{algorithm}[H]
\caption{MitM Detection, Warning, and Warning Persistence}\label{alg:detect}
\KwData{$\mathit{evidence}[]\gets[\,]$; per conversation, $\mathit{PRK}$ as extracted in Alg.~\ref{alg:dualindex}}
\Fn{\normalfont\textsf{DetectMitM}($\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}},\mathit{expected\_pk}$)}{
  $pk\gets \textsf{RetrieveKey}(\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
  \If{$pk\ne\mathit{expected\_pk}$}{
    $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
    $\textsf{WriteWarning}(\mathit{server},\lwarn,(\texttt{"mismatch"},pk,\mathit{ts}),\mathit{PRK})$\;
    \textbf{raise} SecurityAlert\;
  }
  \KwRet $pk$\;
}
\Fn{\normalfont\textsf{RecordWarning}($\lwarn$)}{
  $(v,\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}},\mathit{sth}_w)\gets \mathit{server}.\textsf{Search}(\lwarn)$;\ \ \Req $v\ne\bot$\;
  \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth}_w)$ and the chain $(\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}})$ verifies against $\mathit{sth}_w$ \tcp*{evidence is retained only once it verifies}
  $\mathit{evidence}.\textsf{append}((\lwarn,\mathit{sth}_w,\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}}))$ \tcp*{inclusion proof at the committing epoch}
}
\Fn{\normalfont\textsf{PersistenceCheck}($\mathit{server}$)}{
  \ForEach{$(\lwarn,\mathit{sth}_w,\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}})\in \mathit{evidence}$}{
    $v\gets \textsf{VerifiedLookup}(\mathit{server},\lwarn,\mathit{last\_sth})$\;
    \If{$v=\bot$}{
      \tcc{a committed leaf has disappeared: inclusion under $\mathit{sth}_w$, proven non-inclusion
           under $\mathit{last\_sth}$, and the consistency proof between them are non-repudiable,
           publishable proof of deletion (\cref{ssec:design-detection})}
      $\pi_{\mathrm{cons}}\gets \mathit{server}.\textsf{ConsistencyProof}(\mathit{sth}_w.\mathit{size},\mathit{last\_sth}.\mathit{size})$\;
      \textbf{raise} DeletionEvidence$(\lwarn,\mathit{sth}_w,\pi_{\mathrm{prefix}},\pi_{\mathrm{incl}},\mathit{last\_sth},\pi_{\mathrm{cons}})$\;
    }
  }
}
\Fn{\normalfont\textsf{BackgroundMonitor}($\mathit{server},\mathit{conversations}$)}{
  \ForEach{$(\sharedS,\mathit{id_{self}},\mathit{id_{peer}},\mathit{own\_pk})\in \mathit{conversations}$}{
    $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
    $\textsf{WindowedCatchUp}(\mathit{server},\lself,\lwarn,\mathit{PRK})$\;
    $v\gets \textsf{VerifiedLookup}(\mathit{server},\lself,\mathit{last\_sth})$\;
    \If{$v\ne\mathit{own\_pk}$}{
      $\textsf{WriteWarning}(\mathit{server},\lwarn,(\texttt{"key-replaced"},v,\mathit{ts}),\mathit{PRK})$\;
      \textbf{raise} SecurityAlert\;
    }
  }
  $\textsf{PersistenceCheck}(\mathit{server})$ \tcp*{written warnings must remain committed}
}
\end{algorithm}

\paragraph{The auditor (Algorithm~\ref{alg:auditor}).} The auditor runs against the same networked
key-transparency server the clients use, from an independent vantage point. It is not an in-process
copy of the server's state but an external verifier that fetches signed heads and per-epoch deltas
over the network. It maintains its own mirror of the committed directory, reconstructed by replaying
each epoch's declared delta from epoch~0, and checks epoch~$e$'s declared root against that
reconstruction rather than trusting server-supplied per-operation membership proofs, after the manner
of the IETF KeyTrans \textsf{AuditorUpdate}~\cite{ietf-keytrans-protocol}. Each epoch is replayed onto
a scratch clone of the mirror, and the live mirror, the audited-epoch cursor, and the baseline head are
advanced as a single all-or-nothing transaction only if the entire head passes every stage; an
epoch-delta fetch failure aborts the transaction and is retried on the next poll rather than latching a fork alarm
(matching the implementation's \texttt{transition\_complete} guard). \Cref{alg:auditor} states this
whole-head boundary as realized by the networked auditor binary, which stages the entire epoch range
and commits the mirror, the cursor, and the baseline head together; the in-process library auditor
used by the unit tests is an incremental variant that advances the mirror and cursor after each
verified epoch (so only already-verified intermediate states can persist) and appends the head last,
which is behaviourally equivalent on an honest log and never commits an unverified epoch. Because
every leaf commits its
mutable or immutable kind into its hash, the transition audit distinguishes the two families: a
rotation is accepted only on a mutable (global-VRF identity) leaf and rejected on an immutable (raw
dual-index, CEA, or warning) leaf, as is any rotation that misdeclares the kind. An insert is
admitted only against an absent label, so an overwrite cannot masquerade as an insert, and the
replayed root must equal the epoch's declared root, so any divergence between a declared delta and
its committed effect alarms. This is what
structurally rejects a post-compromise overwrite of a committed CEA fingerprint that a receiver's
per-epoch verdict alone cannot catch (\cref{ssec:design-detection}).

\Cref{alg:auditor} specifies the \emph{networked} auditor binary, which keeps a single trusted
baseline head $\mathit{last}$ and, once replay succeeds, runs \textsf{DetectFork} against one current
second-vantage head. \textsf{AuditEpoch} classifies the new head against the baseline (First,
Regression, SameSizeForked, Grew with a consistency proof, or an Idempotent re-poll), requests a
consistency proof only on growth, and propagates a verdict: it returns \textsc{Unavailable} when the
primary head cannot be fetched or parsed, alarms and returns \textsc{Malformed}, \textsc{Rollback},
\textsc{Fork}, or \textsc{TransitionFailed} on the corresponding failure, holds
without committing on an \textsc{Incomplete} epoch-delta fetch, and otherwise commits the mirror,
cursor, and baseline as one transaction. After committing it distinguishes \textsc{Consistent} (a valid
second vantage was available and agreed) from \textsc{ConsistentNoVantage} (no valid second-vantage head
was available, so the primary chain advanced but no G4 verdict was obtained), so a caller can never read
a missing cross-vantage check as a passed one.
\textsf{DetectFork} first requires the second-vantage head to be present, validly signed, \emph{and}
well-formed; if it is not, it returns \textsc{InvalidHead}, an availability or transport fault that
yields no equivocation verdict and, because $\textsc{InvalidHead}\ne\textsc{Fork}$, does not block the
primary commit (G4 detection needs an honest second vantage, \cref{ssec:eval-auditor}). Given a valid
second-vantage head it returns \textsc{Fork} when the two heads share a size but differ in root, or a
consistency proof between the two vantages is refused or fails, and \textsc{Consistent} otherwise. The
in-process library auditor (used by the unit tests) implements only the pure head-comparison half: it
compares an incoming second-vantage head against every recorded head and reports a fork on a same-epoch
root or size disagreement, an equal-size different-root pair, or a later-but-smaller rollback, but it
does \emph{not} itself fetch a between-size consistency proof; the networked binary adds exactly that
step, fetching and verifying the consistency proof between the two vantage sizes.

\begin{algorithm}[H]
\caption{Networked auditor (fork detection and per-epoch transition audit)}\label{alg:auditor}
\KwData{trusted baseline head $\mathit{last}$ (initially $\bot$); committed $\mathit{mirror}$ (a PrefixTree,
empty initially); audited-epoch cursor $\mathit{audited\_epochs}\gets0$; trusted STH key $pk_{\mathrm{sth}}$}
\Fn{\normalfont\textsf{AuditEpoch}($\mathit{server},\mathit{peer}$)}{
  $\mathit{sth_{new}}\gets \mathit{server}.\textsf{GetCurrentSTH}()$\;
  \lIf{$\mathit{sth_{new}}=\bot$}{\KwRet \textsc{Unavailable} \tcp*[f]{fetch/parse failed: retryable, state unchanged}}
  \lIf{$\neg\,\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth_{new}})\ \vee\ \mathit{sth_{new}}.\mathit{epoch}+1\ne \mathit{sth_{new}}.\mathit{size}$}{\textbf{alarm};\ \KwRet \textsc{Malformed}}
  \If{$\mathit{last}\ne\bot$}{
    \lIf{$\mathit{sth_{new}}.\mathit{epoch}<\mathit{last}.\mathit{epoch}\ \vee\ \mathit{sth_{new}}.\mathit{size}<\mathit{last}.\mathit{size}$}{\textbf{alarm};\ \KwRet \textsc{Rollback}}
    \lIf{$\mathit{sth_{new}}.\mathit{size}=\mathit{last}.\mathit{size}\ \wedge\ \mathit{sth_{new}}.\mathit{root}\ne\mathit{last}.\mathit{root}$}{\textbf{alarm};\ \KwRet \textsc{Fork}}
    \If{$\mathit{sth_{new}}.\mathit{size}>\mathit{last}.\mathit{size}$}{
      $\pi\gets \mathit{server}.\textsf{ConsistencyProof}(\mathit{last}.\mathit{size},\mathit{sth_{new}}.\mathit{size})$\;
      \lIf{$\neg\,\textsf{VerifyConsistency}(\mathit{last}.\mathit{size},\mathit{sth_{new}}.\mathit{size},\mathit{last}.\mathit{root},\mathit{sth_{new}}.\mathit{root},\pi)$}{\textbf{alarm};\ \KwRet \textsc{Fork}}
    }
  }
  $\mathit{scratch}\gets \textsf{clone}(\mathit{mirror})$;\ \ $\mathit{cursor}\gets \mathit{audited\_epochs}$;\ \ $\mathit{complete}\gets \textbf{true}$\;
  \ForEach{not-yet-audited epoch $e\le \mathit{sth_{new}}.\mathit{epoch}$ \emph{(including $e=0$)}}{
    $\mathit{delta}_e\gets \mathit{server}.\textsf{GetEpochDelta}(e,\mathit{sth_{new}}.\mathit{size})$ \tcp*{delta pinned to the signed tree size}
    \lIf{$\mathit{delta}_e = \bot$}{$\mathit{complete}\gets \textbf{false}$;\ \textbf{break} \tcp*[f]{fetch failure: retryable}}
    \lIf{$\textsf{AuditTransition}(\mathit{scratch},e,\mathit{delta}_e,\mathit{sth_{new}}.\mathit{root},\mathit{sth_{new}}.\mathit{size})=\textsc{TransitionFailed}$}{\textbf{alarm};\ \KwRet \textsc{TransitionFailed}}
    $\mathit{cursor}\gets \mathit{cursor}+1$\;
  }
  \lIf{$\neg\,\mathit{complete}$}{\KwRet \textsc{Incomplete}}
  $\mathit{vb}\gets\textsf{DetectFork}(\mathit{peer}.\textsf{CurrentHead}(),\mathit{sth_{new}},\mathit{server})$\;
  \lIf{$\mathit{vb}=\textsc{Fork}$}{\textbf{alarm};\ \KwRet \textsc{Fork}}
  $\mathit{mirror}\gets \mathit{scratch}$;\ \ $\mathit{audited\_epochs}\gets \mathit{cursor}$;\ \ $\mathit{last}\gets \mathit{sth_{new}}$ \tcp*{atomic commit}
  \lIf{$\mathit{vb}=\textsc{InvalidHead}$}{\KwRet \textsc{ConsistentNoVantage} \tcp*[f]{committed; no G4 verdict}}
  \KwRet \textsc{Consistent}\;
}
\Fn{\normalfont\textsf{AuditTransition}($\mathit{scratch},e,\mathit{delta}_e,r_{\mathrm{head}},\mathit{size}_{\mathrm{head}}$)}{
  $(r_{e},\mathit{ops}_e,\pi_{\mathrm{incl}})\gets \mathit{delta}_e$\;
  \lIf{$\pi_{\mathrm{incl}}$ does not prove leaf index $e$ with value $r_{e}$ included under $r_{\mathrm{head}}$ at tree size $\mathit{size}_{\mathrm{head}}$}{\KwRet \textsc{TransitionFailed}}
  \ForEach{operation $o\in \mathit{ops}_e$ \emph{in declared order}}{
    \uIf{$o=\textsf{Insert}(\ell,\mathit{kind},v)$}{\lIf{$\mathit{scratch}.\textsf{get}(\ell)\ne\bot$}{\KwRet \textsc{TransitionFailed}} \textbf{else} insert $(\ell,\mathit{kind},v)$\;}
    \Else{$o=\textsf{Rotate}(\ell,\mathit{kind},v_{\mathrm{old}},v)$: \lIf{$\neg(\mathit{kind}=\textsc{mutable}\wedge \mathit{scratch}.\textsf{kind}(\ell)=\mathit{kind}\wedge \mathit{scratch}.\textsf{get}(\ell)=v_{\mathrm{old}})$}{\KwRet \textsc{TransitionFailed}} \textbf{else} set $\ell\mapsto v$\;}
  }
  \lIf{$\mathit{scratch}.\textsf{root}()\ne r_{e}$}{\KwRet \textsc{TransitionFailed}}
  \KwRet \textsc{Pass}\;
}
\Fn{\normalfont\textsf{DetectFork}($\mathit{sth_b},\mathit{sth_{new}},\mathit{server}$)}{
  \lIf{$\mathit{sth_b}=\bot\ \vee\ \neg\,\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth_b})\ \vee\ \mathit{sth_b}.\mathit{epoch}+1\ne\mathit{sth_b}.\mathit{size}$}{\KwRet \textsc{InvalidHead} \tcp*[f]{no G4 verdict; does not block the primary commit}}
  \uIf{$\mathit{sth_b}.\mathit{size}=\mathit{sth_{new}}.\mathit{size}$}{
    \lIf{$\mathit{sth_b}.\mathit{root}\ne\mathit{sth_{new}}.\mathit{root}$}{\KwRet \textsc{Fork}}
  }
  \Else{
    $m,n\gets$ smaller, larger of $\mathit{sth_b}.\mathit{size}$ and $\mathit{sth_{new}}.\mathit{size}$\;
    $\pi\gets \mathit{server}.\textsf{ConsistencyProof}(m,n)$\;
    \lIf{refused $\vee\ \neg\,\textsf{VerifyConsistency}(m,n,\ \text{signed roots at }m,n,\ \pi)$}{\KwRet \textsc{Fork}}
  }
  \KwRet \textsc{Consistent}\;
}
\end{algorithm}

\paragraph{Continuous mode (Algorithm~\ref{alg:cea}).} CEA authenticates the Double Ratchet epoch by
epoch, following the ACKA CEA construction. It folds each epoch's secret into a chained key $k$ and
publishes a directional fingerprint over the new ratchet public key, which the receiver recomputes and
compares. A mismatch shows that the committed fingerprint is inconsistent with the receiver's chained state and
authenticated ratchet public key $T_t$; the verifier reports this as a fork without attributing the
divergence to a particular input. Because the committed
fingerprint is immutable, an operator that later overwrites it is caught by the auditor's transition
audit rather than by the receiver's per-epoch verdict. Labels use the canonical $\min/\max$ identity
order, so both parties address one slot per epoch, and the fingerprint is directional
(sender-then-receiver) and computed over the current, pre-update state. On registration the sender
first checks that its label is unoccupied, since an occupied label signals impersonation, and advances
its chain only after confirming that the write committed. The receiver advances only on a
\textsc{Consistent} verdict, leaving its state unchanged on \textsc{Fork} or \textsc{NotPosted}.

\begin{algorithm}[H]
\caption{Continuous Entity Authentication mode (CEA; per-ratchet, ACKA CEA construction)}\label{alg:cea}
\KwData{per conversation: chained secret $k$; epoch $t$; warning label $\lwarn$ and $\mathit{PRK}$
from Alg.~\ref{alg:dualindex} (the label uses canonical $\min/\max$ id order: one slot per epoch;
the fingerprint is directional, sender-then-receiver, \cref{ssec:design-modes})}
\Fn{\normalfont\textsf{CeaSetup}($\mathit{pss},\mathit{id_A},\mathit{id_B}$)}{
  $k\gets \KDF(\mathit{pss},\min(\mathit{id_A},\mathit{id_B})\concat \max(\mathit{id_A},\mathit{id_B}))$;\ \ $t\gets 0$\;
}
\Fn{\normalfont\textsf{CeaUpdate}($I$)}{
  $k\gets \KDF(k,\KDF(I,\texttt{"CEA"}))$;\ \ $t\gets t+1$\;
}
\Fn{\normalfont\textsf{CeaLprint}($\mathit{id_A},\mathit{id_B}$)}{
  \KwRet $\Hash(k\concat \min(\mathit{id_A},\mathit{id_B})\concat \max(\mathit{id_A},\mathit{id_B})\concat t)$\;
}
\Fn{\normalfont\textsf{CeaFprint}($\mathit{id_{sender}},\mathit{id_{peer}},T_t$)}{
  $\mathit{fkey}\gets \KDF(k,\mathit{id_{sender}}\concat \mathit{id_{peer}}\concat t)$\;
  \KwRet $\Hash(\mathit{fkey}\concat T_t)$\;
}
\Fn{\normalfont\textsf{CeaRegister}($\mathit{server},\mathit{id_{self}},\mathit{id_{peer}},T_t,I,\mathit{sth}$)}{
  $L\gets \textsf{CeaLprint}(\mathit{id_{self}},\mathit{id_{peer}})$;\ \ $F\gets \textsf{CeaFprint}(\mathit{id_{self}},\mathit{id_{peer}},T_t)$\;
  \If{$\textsf{VerifiedLookup}(\mathit{server},L,\mathit{sth})\ne\bot$}{
    \textbf{raise} SecurityAlert\;
  }
  $\mathit{server}.\textsf{UpdateRaw}(L,F)$;\ \ $\mathit{sth'}\gets \mathit{server}.\textsf{CommitEpoch}()$\;
  \Req $\textsf{VerifiedLookup}(\mathit{server},L,\mathit{sth'})=F$\;
  $\textsf{CeaUpdate}(I)$ \tcp*{advance after the confirmed write}
}
\Fn{\normalfont\textsf{CeaVerify}($\mathit{server},\mathit{id_{sender}},\mathit{id_{self}},T_t,I,\lwarn,\mathit{sth}$)}{
  $L\gets \textsf{CeaLprint}(\mathit{id_{sender}},\mathit{id_{self}})$;\ \ $F^{*}\gets \textsf{CeaFprint}(\mathit{id_{sender}},\mathit{id_{self}},T_t)$\;
  $v\gets \textsf{VerifiedLookup}(\mathit{server},L,\mathit{sth})$\;
  \lIf{$v=\bot$}{\KwRet \textsc{NotPosted} \tcp*[f]{no advance}}
  \If{$v\ne F^{*}$}{
    $\textsf{WriteWarning}(\mathit{server},\lwarn,(\texttt{"ratchet-fork"},v,t),\mathit{PRK})$\;
    \KwRet \textsc{Fork} \tcp*{no advance}
  }
  $\textsf{CeaUpdate}(I)$ \tcp*{advance on \textsc{Consistent}}
  \KwRet \textsc{Consistent}\;
}
\end{algorithm}
