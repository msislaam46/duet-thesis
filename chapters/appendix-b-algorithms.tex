This appendix specifies the eight algorithms the thesis refers to. They fall into two groups.
Algorithms~\ref{alg:merklelog}--\ref{alg:dualindex} are the data-structure and derivation
primitives: the Merkle log tree, the sparse prefix tree, the directory server that combines
them, and the dual-index label derivation. Algorithms~\ref{alg:client}--\ref{alg:cea} are the
protocol procedures built on those primitives: client verification, MitM detection and warning
persistence, the auditor, and the continuous-authentication mode. \Cref{tab:alg-purpose} states the
purpose of each.

\paragraph{Notation.} $\Hash(\cdot)$ is SHA-256. Tree hashing is domain-separated per RFC 6962/9162~\cite{rfc6962,rfc9162}: a
leaf hashes as $\Hleaf(x)=\Hash(\texttt{0x00}\concat x)$ and an internal node as
$\Hnode(l,r)=\Hash(\texttt{0x01}\concat l\concat r)$. The operator $\concat$ denotes byte
concatenation. Identities are canonical 16-byte account identifiers. Their fixed width makes the
$\mathit{id_{self}}\concat\mathit{id_{peer}}$ concatenation unambiguous without length prefixes
(Algorithm~\ref{alg:dualindex}). Both the \texttt{duet} crate and client component enforce this
requirement before label derivation.

\paragraph{The Merkle log (Algorithm~\ref{alg:merklelog}).} The Merkle log is the append-only layer beneath every
signed head. It fixes the RFC~9162~\cite{rfc9162} tree-hash and proof conventions used everywhere else: inclusion
proofs bind one leaf to a root, and consistency proofs bind an old root to a newer one, the
primitive that windowed catch-up (\cref{sec:design-windowed}) and the auditor's fork check both
build on. The $m=0$ case is accepted as the local trust-on-first-use convention.
\begin{algorithm}[H]
\caption{Merkle Log Tree (RFC 9162)}\label{alg:merklelog}
\KwData{$\mathit{leaves}[]\gets[\,]$ (the ordered leaf list, written $D$ in the RFC~9162 slice notation, so $D=\mathit{leaves}$)}
\Fn{\normalfont\textsf{Append}($\mathit{record}$)}{
  $\mathit{leaves}.\textsf{append}(\Hleaf(\mathit{record}))$\;
  \KwRet $|\mathit{leaves}|-1$ \tcp*{index of new leaf}
}
\Fn{\normalfont\textsf{MTH}($D[a{:}b]$)}{
  $n\gets b-a$\;
  \lIf{$n=0$}{\KwRet $\Hash(\varepsilon)$}
  \lIf{$n=1$}{\KwRet $\mathit{leaves}[a]$}
  $k\gets$ largest power of two $<n$ \tcp*{RFC split point}
  \KwRet $\Hnode(\textsf{MTH}(D[a{:}a{+}k]),\textsf{MTH}(D[a{+}k{:}b]))$\;
}
\Fn{\normalfont\textsf{Root}($\mathit{tree\_size}$)}{
  \KwRet $\textsf{MTH}(D[0{:}\mathit{tree\_size}])$\;
}
\Fn{\normalfont\textsf{InclusionProof}($m,\mathit{tree\_size}$)}{
  \Req $0\le m<\mathit{tree\_size}$\;
  $\mathit{path}\gets$ sibling subtree hashes along RFC splits, leaf-to-root\;
  \KwRet $\mathit{path}$\;
}
\Fn{\normalfont\textsf{VerifyInclusion}($\mathit{record},m,\mathit{tree\_size},\mathit{root},\mathit{path}$)}{
  \Req $0\le m<\mathit{tree\_size}$\;
  $\mathit{fn}\gets m$;\ \ $\mathit{sn}\gets \mathit{tree\_size}-1$;\ \ $r\gets \Hleaf(\mathit{record})$\;
  \ForEach{$p\in\mathit{path}$}{
    \Req $\mathit{sn}\ne 0$\;
    \uIf{$\mathit{fn}$ odd $\vee\ \mathit{fn}=\mathit{sn}$}{
      $r\gets \Hnode(p,r)$\;
      \lIf{$\mathit{fn}$ even}{right-shift $\mathit{fn},\mathit{sn}$ until $\mathit{fn}$ odd $\vee\ \mathit{fn}=0$}
    }
    \lElse{$r\gets \Hnode(r,p)$}
    $\mathit{fn}\gets\lfloor \mathit{fn}/2\rfloor$;\ \ $\mathit{sn}\gets\lfloor \mathit{sn}/2\rfloor$\;
  }
  \KwRet $\mathit{sn}=0\wedge r=\mathit{root}$ \tcp*{rejects short/padded paths}
}
\Fn{\normalfont\textsf{ConsistencyProof}($m,n$)}{
  \Req $0\le m\le n$ \tcp*{$m=0$ accepted as the local TOFU convention, while RFC 9162 assumes $0<m<n$}
  \lIf{$m=0\vee m=n$}{\KwRet $[\,]$}
  \KwRet $\textsf{SUBPROOF}(m,D[0{:}n],\mathrm{true})$ \tcp*{RFC 9162 Section 2.1.4.1}
}
\Fn{\normalfont\textsf{VerifyConsistency}($m,n,\mathit{old\_root},\mathit{new\_root},\mathit{proof}$)}{
  \lIf{$m=n$}{\KwRet $\mathit{old\_root}=\mathit{new\_root}\wedge \mathit{proof}=[\,]$}
  \lIf{$m=0$}{\KwRet $\mathit{proof}=[\,]$}
  \Req $\mathit{proof}\ne[\,]$\;
  \lIf{$m$ is a power of two}{$\mathit{proof}\gets \mathit{old\_root}:\mathit{proof}$}
  $\mathit{fn}\gets m-1$;\ \ $\mathit{sn}\gets n-1$\;
  right-shift $\mathit{fn},\mathit{sn}$ while $\mathit{fn}$ odd\;
  $\mathit{fr}\gets\mathit{proof}[0]$;\ \ $\mathit{sr}\gets\mathit{proof}[0]$ \tcp*{both roots intentionally seed from the same $\mathit{proof}[0]$ (RFC 9162 Section~2.1.4.2)}
  \ForEach{$c\in\mathit{proof}[1{:}]$}{
    \Req $\mathit{sn}\ne 0$\;
    \uIf{$\mathit{fn}$ odd $\vee\ \mathit{fn}=\mathit{sn}$}{
      $\mathit{fr}\gets\Hnode(c,\mathit{fr})$;\ \ $\mathit{sr}\gets\Hnode(c,\mathit{sr})$\;
      \lIf{$\mathit{fn}$ even}{right-shift $\mathit{fn},\mathit{sn}$ until $\mathit{fn}$ odd $\vee\ \mathit{fn}=0$}
    }
    \lElse{$\mathit{sr}\gets\Hnode(\mathit{sr},c)$}
    $\mathit{fn}\gets\lfloor \mathit{fn}/2\rfloor$;\ \ $\mathit{sn}\gets\lfloor \mathit{sn}/2\rfloor$\;
  }
  \KwRet $\mathit{fr}=\mathit{old\_root}\wedge \mathit{sr}=\mathit{new\_root}\wedge \mathit{sn}=0$\;
}
\end{algorithm}

\paragraph{The prefix tree (Algorithm~\ref{alg:prefix}).} The prefix tree is the label-keyed
authenticated dictionary holding the directory's current state. Its underlying structure is a
path-compressed binary trie over 256-bit labels, and each leaf commits its mutable-or-immutable kind
into the hash. Lookups return compressed inclusion and non-inclusion proofs. A verified
non-inclusion proof establishes that the warning slot is empty.

\begin{algorithm}[H]
\caption{Sparse Binary Merkle Prefix Tree (label-keyed)}\label{alg:prefix}
\KwData{path-compressed binary trie over 256-bit labels; empty hash $e=\Hash(\varepsilon)$}
\Fn{\normalfont\textsf{Insert}($\mathit{label},\mathit{value},\mathit{kind}$)}{
  \Req $|\mathit{label}|=32\ \wedge\ \mathit{kind}\in\{\textsc{mutable},\textsc{immutable}\}$ \tcp*{both enforced, kind bound into the leaf}
  place leaf $\Hleaf(\mathit{kind}\concat\mathit{label}\concat\mathit{value})$ at the bit-path of $\mathit{label}$\;
  recompute hashes along the insertion path only \tcp*{$O(d)$, $d=256$}
  \KwRet \textsf{RootHash}()\;
}
\Fn{\normalfont\textsf{Lookup}($\mathit{label}$)}{
  walk the bit-path of $\mathit{label}$\;
  $\pprefix\gets(\mathit{label},\mathit{value}\text{ or }\bot,\mathit{kind},\mathit{bitmap},\mathit{siblings})$ \tcp*{compressed proof}
  \KwRet $(\mathit{value}\text{ or }\bot,\ \pprefix)$\;
}
\Fn{\normalfont\textsf{RootFromProof}($\pprefix$)}{
  \lIf{$|\pprefix.\mathit{label}|\ne 32\ \vee\ |\pprefix.\mathit{bitmap}|\ne 32$}{\KwRet $\bot$ \tcp*[f]{malformed proof: reject before hashing}}
  \uIf{$\pprefix.\mathit{value}=\bot$}{$c\gets e$ \tcp*[f]{absence: leaf hash is the empty hash}}
  \uElseIf{$\pprefix.\mathit{kind}\notin\{\textsc{mutable},\textsc{immutable}\}$}{\KwRet $\bot$ \tcp*[f]{present leaf with an out-of-domain kind}}
  \Else{$c\gets\Hleaf(\pprefix.\mathit{kind}\concat\pprefix.\mathit{label}\concat\pprefix.\mathit{value})$ \tcp*[f]{kind authenticated by the root}}
  \For{$i\gets 0$ \KwTo $255$}{
    $s\gets$ next sibling from $\pprefix.\mathit{siblings}$ if $\pprefix.\mathit{bitmap}[i]$ set, else $e$\;
    \lIf{$c=e\wedge s=e$}{\textbf{continue}}
    $c\gets \Hnode(c,s)$ or $\Hnode(s,c)$ per bit $255-i$ of $\pprefix.\mathit{label}$\;
  }
  \Req all siblings consumed, else \KwRet $\bot$\;
  \KwRet $c$\;
}
\Fn{\normalfont\textsf{VerifyProof}($\pprefix,\mathit{expected\_root}$)}{
  \KwRet $\textsf{RootFromProof}(\pprefix)=\mathit{expected\_root}$\;
}
\end{algorithm}

\paragraph{The server (Algorithm~\ref{alg:server}).} The directory server composes the two structures into the
key-transparency service: raw-label writes are write-once, \textsf{CommitEpoch} appends the prefix
root to the log and signs the head (binding epoch, size, root, timestamp, and the configuration
commitment), and \textsf{Search} answers with the value plus the full proof chain a verifier needs.
\textsf{SearchByUserId} adds the VRF step (ECVRF, RFC~9381~\cite{rfc9381}) that derives global labels through the server's ECVRF evaluation.
\begin{algorithm}[H]
\caption{Key-Transparency Directory Server}\label{alg:server}
\KwData{$\mathit{log}\gets$ MerkleLogTree, $\mathit{live}\gets$ PrefixTree, $\mathit{committed}\gets\bot$, $\mathit{committed\_prefix\_roots}\gets[\,]$, $\mathit{sth}_{\mathrm{latest}}\gets\bot$, $\mathit{epoch}\gets0$, $(sk_{\mathrm{vrf}},pk_{\mathrm{vrf}})$, $(sk_{\mathrm{sth}},pk_{\mathrm{sth}})$, config $\mathit{suite}$, $\mathit{version}$, $\mathit{epoch\_interval}$, $\mathit{raw\_label\_policy}$}
\Fn{\normalfont\textsf{UpdateByUserId}($\mathit{user\_id},pk$)}{
  $(\beta,\pvrf)\gets \mathsf{VRF.Prove}(sk_{\mathrm{vrf}},\mathit{user\_id})$ \tcp*{ECVRF, RFC 9381}
  $\mathit{label}\gets\beta[0{:}32]$ \tcp*{32-byte truncation}
  $\mathit{old}\gets\mathit{live}.\textsf{Lookup}(\mathit{label}).\mathit{value}$ \tcp*{the previous value decides insert vs.\ rotation}
  $\mathit{live}.\textsf{Insert}(\mathit{label},pk,\textsc{mutable})$ \tcp*{leaf $\Hleaf(\textsc{mutable}\concat\mathit{label}\concat pk)$}
  \eIf{$\mathit{old}=\bot$}{record $\textsf{Insert}(\mathit{label},\textsc{mutable},pk)$ in the declared delta\;}{record $\textsf{Rotate}(\mathit{label},\textsc{mutable},\mathit{old},pk)$ in the declared delta\;}
  \KwRet $(\mathit{label},\pvrf)$\;
}
\Fn{\normalfont\textsf{UpdateRaw}($\mathit{label},\mathit{value}$)}{
  \Req $\mathit{live}.\textsf{Lookup}(\mathit{label}).\mathit{value}=\bot$ \tcp*{write-once: occupied labels rejected (\cref{ssec:design-detection})}
  $\mathit{live}.\textsf{Insert}(\mathit{label},\mathit{value},\textsc{immutable})$ \tcp*{dual-index/CEA leaf $\Hleaf(\textsc{immutable}\concat\mathit{label}\concat\mathit{value})$}
  record \textsf{Insert}$(\mathit{label},\textsc{immutable},\mathit{value})$ in the epoch's declared delta\;
}
\Fn{\normalfont\textsf{CommitEpoch}()}{
  $e\gets \mathit{epoch}$ \tcp*{capture the epoch being committed}
  $r\gets \mathit{live}.\textsf{RootHash}()$\;
  $i\gets \mathit{log}.\textsf{Append}(r)$ \tcp*{prefix root is a log leaf}
  $\mathit{committed\_prefix\_roots}[e]\gets r$ \tcp*{retain the epoch's frozen directory root, checked by the transition audit (Alg.~\ref{alg:auditor})}
  $\mathit{committed}\gets(\mathit{tree}{:}\,\textsf{snapshot}(\mathit{live}),\ \mathit{leaf\_index}{:}\,i)$ \tcp*{named fields, read below as $\mathit{committed}.\mathit{tree}$ and $\mathit{committed}.\mathit{leaf\_index}$}
  freeze the declared delta for this epoch \tcp*{served to auditors via \textsf{GetEpochDelta} (Alg.~\ref{alg:auditor})}
  $\mathit{config\_hash}\gets \Hash(pk_{\mathrm{vrf}}\concat pk_{\mathrm{sth}}\concat \mathit{suite}\concat \mathit{version}\concat \mathit{epoch\_interval}\concat \mathit{raw\_label\_policy})$ \tcp*{configuration commitment (CONIKS $P$), where $\mathit{raw\_label\_policy}$ binds the write-once/delta policy}
  $\mathit{timestamp}\gets\textsf{CurrentUnixTime}()$\;
  $\sigma\gets \textsf{Sign}(sk_{\mathrm{sth}},\ e\concat \mathit{log}.\mathit{size}\concat \mathit{log}.\textsf{Root}\concat \mathit{timestamp}\concat \mathit{config\_hash})$ \tcp*{88-byte signing message}
  $\mathit{sth}\gets(e,\mathit{log}.\mathit{size},\mathit{log}.\textsf{Root},\mathit{timestamp},\mathit{config\_hash},\sigma)$\;
  $\mathit{sth}_{\mathrm{latest}}\gets \mathit{sth}$ \tcp*{published head served by \textsf{Search}}
  $\mathit{epoch}\gets \mathit{epoch}+1$ \tcp*{advance only after signing this epoch's head}
  \KwRet $\mathit{sth}$\;
}
\Fn{\normalfont\textsf{Search}($\mathit{label}$)}{
  \Req $\mathit{committed}\ne\bot$\;
  $(v,\pprefix)\gets \mathit{committed}.\mathit{tree}.\textsf{Lookup}(\mathit{label})$\;
  $\pincl\gets \mathit{log}.\textsf{InclusionProof}(\mathit{committed}.\mathit{leaf\_index},\mathit{log}.\mathit{size})$\;
  \KwRet $(v,\pprefix,\pincl,\mathit{sth}_{\mathrm{latest}})$\;
}
\Fn{\normalfont\textsf{SearchByUserId}($\mathit{user\_id}$)}{
  $(\beta,\pvrf)\gets \mathsf{VRF.Prove}(sk_{\mathrm{vrf}},\mathit{user\_id})$\;
  \KwRet $\textsf{Search}(\beta[0{:}32])$ with $\pvrf$ attached\;
}
\end{algorithm}

\paragraph{Dual-index labels and warnings (Algorithm~\ref{alg:dualindex}).} The per-conversation
label layer of \cref{sec:design-dual}: from each conversation's shared secret, HKDF~\cite{rfc5869} derives the two
party-specific lookup labels and the canonically ordered shared warning label. \textsf{WriteWarning}
publishes a MAC-authenticated warning payload and trusts neither an apparent success nor a write-once
conflict, confirming publication by a verified read-back. \textsf{CheckWarning} raises
\textsc{Alert} on any proven non-empty slot.
\begin{algorithm}[H]
\caption{Dual-Index Label Derivation (HKDF-SHA256, RFC 5869)}\label{alg:dualindex}
\KwIn{shared secret $\sharedS$ from X3DH or PQXDH; identities $\mathit{id_{self}},\mathit{id_{peer}}$ (fixed-width). The warning-MAC key is derived separately by \textsf{DeriveWarnMacKey}, which orders the identities canonically}
\Fn{\normalfont\textsf{DeriveLabels}($\sharedS,\mathit{id_{self}},\mathit{id_{peer}}$)}{
  $\mathit{PRK}\gets \textsf{HKDF-Extract}(\text{salt}=\texttt{"duet-dual-index"},\ \sharedS)$\;
  $\lself\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"lookup"}\concat \mathit{id_{self}}\concat \mathit{id_{peer}},32)$\;
  $\lpeer\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"lookup"}\concat \mathit{id_{peer}}\concat \mathit{id_{self}},32)$\;
  $\lwarn\gets \textsf{HKDF-Expand}(\mathit{PRK},\texttt{"warning"}\concat \min(\mathit{id_{self}},\mathit{id_{peer}})\concat \max(\mathit{id_{self}},\mathit{id_{peer}}),32)$ \tcp*{canonical order (warning only), the same $\min/\max$ used by \textsf{DeriveWarnMacKey}}
  \KwRet $(\lself,\lpeer,\lwarn)$\;
}
\Fn{\normalfont\textsf{DeriveWarnMacKey}($\sharedS,\mathit{id_{self}},\mathit{id_{peer}}$)}{
  $\mathit{PRK}\gets \textsf{HKDF-Extract}(\text{salt}=\texttt{"duet-dual-index"},\ \sharedS)$\;
  \KwRet $\textsf{HKDF-Expand}(\mathit{PRK},\texttt{"warn-mac"}\concat\min(\mathit{id_{self}},\mathit{id_{peer}})\concat\max(\mathit{id_{self}},\mathit{id_{peer}}),32)$ \tcp*{canonical order, both parties derive the same key}
}
\Fn{\normalfont\textsf{WriteWarning}($\mathit{server},\lwarn,\mathit{alert},k_{\mathrm{mac}}$)}{
  \uIf{$\mathit{server}.\textsf{UpdateRaw}(\lwarn,\ \mathit{alert}\concat\textsf{MAC}(k_{\mathrm{mac}},\mathit{alert}))$ succeeds}{
    $\mathit{server}.\textsf{CommitEpoch}()$ \tcp*{commit the successful write}
  }
  \uElseIf{\normalfont write-once conflict (typed 409)}{
    \emph{skip} \textsf{CommitEpoch} \tcp*{occupied slot, tolerated without trust}
  }
  \Else{
    \textbf{propagate} the error \tcp*{a transport failure is an availability problem, not censorship}
  }
  \lIf{$\textsf{VerifiedLookup}(\mathit{server},\lwarn)$ rejects or returns $\bot$}{\textbf{alarm} \textsc{WarningCensored} \tcp*[f]{no provable committed warning under the fresh returned head (invalid proof or proven absence)}}
}
\Fn{\normalfont\textsf{CheckWarning}($\mathit{server},\lwarn,k_{\mathrm{mac}}$)}{
  $v\gets \textsf{VerifiedLookup}(\mathit{server},\lwarn)$ \tcp*{proof required (Alg.~\ref{alg:client})}
  \lIf{$v=\bot$}{\KwRet \textsc{Safe}}
  \lIf{$\textsf{MAC-Verify}(k_{\mathrm{mac}},v)$}{\KwRet \textsc{Alert} \tcp*[f]{valid pair-bound MAC from a shared-key holder}}
  \KwRet \textsc{Alert} \tcp*{an unauthenticated value at the secret-derived slot still produces an alert}
}
\end{algorithm}

\paragraph{Client verification (Algorithm~\ref{alg:client}).} These procedures define the checks
applied before the client accepts a directory value.
\textsf{VerifiedLookup} returns a value only after every check in
Algorithm~\ref{alg:client} succeeds. It verifies the signed head returned with the lookup and folds
it into the client's retained append-only history. Note that a concurrent commit may cause the
returned head to be newer than the retained baseline. The procedure rejects the response if any check fails.

The directory may advance between two monitor lookups. The Signal Desktop verifier accepts a newer
signed head after verifying that it extends the client's last trusted head. This links each accepted
lookup to the client's verified append-only history.

\textsf{Register} attempts to place the client's key at its secret-derived owner label. A 409
write-once conflict skips the commit. Registration succeeds only after a verified read-back shows
that the label is occupied, and the background monitor checks the returned value against the
client's local identity key during the same tick. \textsf{RetrieveKey} verifies and returns the
peer-slot value, while \textsf{WindowedCatchUp} verifies the log extension from the client's last
trusted head.

For each active conversation, the background monitor composes these operations and performs
windowed catch-up before checking the owner, warning, and peer slots.

{\SetAlFnt{\footnotesize}\SetInd{0.35em}{0.35em}%
\begin{algorithm}[H]
\caption{Client Verification, Registration, Retrieval, Windowed Catch-Up}\label{alg:client}
\KwData{$\mathit{last\_sth}\gets\bot$; pinned $pk_{\mathrm{vrf}}$, $pk_{\mathrm{sth}}$, $\mathit{suite}$, $\mathit{version}$, $\mathit{epoch\_interval}$, and $\mathit{raw\_label\_policy}$}
\Fn{\normalfont\textsf{VerifiedLookup}($\mathit{server},\mathit{label},[\mathit{user\_id},\pvrf]$)}{
  $(v,\pprefix,\pincl,\mathit{sth'})\gets \mathit{server}.\textsf{Search}(\mathit{label})$ \tcp*{$\mathit{sth'}$ is the head returned with this lookup}
  \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth'})$ \tcp*{Ed25519 over epoch$\concat$tree\_size$\concat$root$\concat$timestamp$\concat$config\_hash}
  $\mathit{expected\_config\_hash}\gets \Hash(pk_{\mathrm{vrf}}\concat pk_{\mathrm{sth}}\concat \mathit{suite}\concat \mathit{version}\concat \mathit{epoch\_interval}\concat \mathit{raw\_label\_policy})$ \tcp*{recompute from the client's pinned parameters}
  \Req $\mathit{sth'}.\mathit{config\_hash}=\mathit{expected\_config\_hash}\ \wedge\ \mathit{sth'}.\mathit{epoch}+1=\mathit{sth'}.\mathit{tree\_size}$ \tcp*{expected configuration commitment + well-formed head}
  \Req $\pprefix.\mathit{label}=\mathit{label}$ \tcp*{label binding: the proof concerns the queried label}
  \If{$\mathit{user\_id}$ given}{
    $\beta\gets \mathsf{VRF.Verify}(pk_{\mathrm{vrf}},\mathit{user\_id},\pvrf)$;\ \ \Req $\beta[0{:}32]=\mathit{label}$ \tcp*[f]{VRF path: derived label equals the queried label}\;
  }
  \Req $\pprefix.\mathit{value}=v$\;
  \lIf{$\mathit{label}$ is a raw (dual-index, warning, CEA) label $\wedge\ v\ne\bot$}{\Req $\pprefix.\mathit{kind}=\textsc{immutable}$ \tcp*[f]{raw path: a present client-derived leaf is write-once}}
  $r\gets \textsf{RootFromProof}(\pprefix)$;\ \ \Req $r\ne\bot$\;
  \Req $\textsf{VerifyInclusion}(r,\pincl.\mathit{leaf\_index},\pincl.\mathit{tree\_size},\mathit{sth'}.\mathit{root},\pincl.\mathit{path})$\;
  \Req $\pincl.\mathit{tree\_size}=\mathit{sth'}.\mathit{tree\_size}\ \wedge\ \pincl.\mathit{leaf\_index}=\mathit{sth'}.\mathit{tree\_size}-1$ \tcp*{current-snapshot binding: require the final log leaf at the signed tree size}
  \Req $\mathit{sth'}$ folds into the client's append-only chain (as in \textsf{WindowedCatchUp}) \tcp*{search-head anchor: require the returned head to extend the retained baseline}
  \KwRet $v$ \tcp*{$v$ may be $\bot$: proven non-inclusion}
}
\Fn{\normalfont\textsf{Register}($\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}},pk_{\mathit{self}}$)}{
  $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
  $\mathit{r}\gets \mathit{server}.\textsf{UpdateRaw}(\lself,pk_{\mathit{self}})$ \tcp*{success, or a typed 409 write-once conflict}
  \uIf{$\mathit{r}$ succeeded}{$\mathit{server}.\textsf{CommitEpoch}()$ \tcp*{fresh write: commit it (prototype only)}}
  \uElseIf{$\mathit{r}$ is not a 409 write-once conflict}{\textbf{raise} $\mathit{r}$ \tcp*{a transport failure propagates}}
  \Else{\tcp{409: the unchanged secret re-derives the occupied label, so confirm by read-back}}
  $v\gets \textsf{VerifiedLookup}(\mathit{server},\lself)$;\ \ \Req $v\ne\bot$ \tcp*{verified non-empty read-back}
  \KwRet $(\lself,\lpeer,\lwarn)$\;
}
\Fn{\normalfont\textsf{RetrieveKey}($\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}}$)}{
  $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
  $\textsf{WindowedCatchUp}(\mathit{server})$ \tcp*{advance the retained append-only baseline}
  $pk\gets \textsf{VerifiedLookup}(\mathit{server},\lpeer)$ \tcp*{$pk=\bot$ is proven non-inclusion, decided by the monitor}
  \KwRet $pk$\;
}
\Fn{\normalfont\textsf{WindowedCatchUp}($\mathit{server}$)}{
  $\mathit{sth_{new}}\gets \mathit{server}.\textsf{GetCurrentSTH}()$;\ \ \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth_{new}})$\;
  $c\gets \Hash(pk_{\mathrm{vrf}}\concat pk_{\mathrm{sth}}\concat \mathit{suite}\concat \mathit{version}\concat \mathit{epoch\_interval}\concat \mathit{raw\_label\_policy})$;\ \ \Req $\mathit{sth_{new}}.\mathit{config\_hash}=c$ \tcp*{configuration commitment: reject an unexpected configuration}
  \Req $\mathit{sth_{new}}.\mathit{epoch}+1=\mathit{sth_{new}}.\mathit{tree\_size}$ \tcp*{reject an epoch/tree-size mismatch before folding}
  \uIf{$\mathit{last\_sth}=\bot$}{
    $\mathit{last\_sth}\gets \mathit{sth_{new}}$ \tcp*{trust-on-first-use anchor (\cref{sec:threat-fair})}
  }\Else{
    \lIf{$\mathit{sth_{new}}.\mathit{epoch}<\mathit{last\_sth}.\mathit{epoch} \lor \mathit{sth_{new}}.\mathit{tree\_size}<\mathit{last\_sth}.\mathit{tree\_size}$}{\textbf{reject} \tcp*[f]{a smaller epoch or tree size is a rollback}}
    \uIf{$\mathit{sth_{new}}.\mathit{tree\_size}=\mathit{last\_sth}.\mathit{tree\_size}$}{
      \Req $\mathit{sth_{new}}.\mathit{root}=\mathit{last\_sth}.\mathit{root}$ \tcp*{equal size with a differing root is same-epoch equivocation}
    }\Else{
      $\pcons\gets \mathit{server}.\textsf{ConsistencyProof}(\mathit{last\_sth}.\mathit{tree\_size},\mathit{sth_{new}}.\mathit{tree\_size})$\;
      \Req $\textsf{VerifyConsistency}(\mathit{last\_sth}.\mathit{tree\_size},\mathit{sth_{new}}.\mathit{tree\_size},\mathit{last\_sth}.\mathit{root},\mathit{sth_{new}}.\mathit{root},\pcons)$\;
      \tcc{one consistency proof covers the complete window}
    }
  }
  $\mathit{last\_sth}\gets \mathit{sth_{new}}$ \tcp*{advance the retained append-only baseline}
}
\end{algorithm}
}

\paragraph{Detection and persistence (Algorithm~\ref{alg:detect}).} Algorithm~\ref{alg:detect} specifies
the monitoring loop that turns a substituted key into an alert. \textsf{CheckPeerKeyAndWarn} compares
the peer key and attempts to publish a warning when a mismatch is found. In the implementation,
\texttt{RemoteKTClient.detectMitm} provides the owner- and warning-slot checks used by the
demonstrations and tests. The live \texttt{monitorConversation} path performs the owner-, warning-, and
peer-slot checks directly.
The shared warning slot is checked \emph{before} the peer slot, so a committed warning takes precedence
over a pending state or an operational \texttt{NotPosted} timeout for an absent peer. A proof-verified
peer-slot absence is reported as pending for the first two checks and as an operational
\texttt{NotPosted} timeout on the third. This fail-closed outcome is distinct from the CEA verdict
\textsc{NotPosted} and does not prove an attack because the peer may not have run the client. A slot
that was present and later becomes absent produces an immediate alert.
\textsf{RecordWarning} and \textsf{PersistenceCheck} retain verified evidence so that a later
deletion of a committed warning becomes publicly verifiable. They are unit-tested as a separate
persistence component and are not integrated into the live background monitor. The live monitor
performs detection and attempts to publish a warning. An unreachable directory ends the global check
or stops the remaining conversation sweep.

{\SetAlFnt{\footnotesize}\SetInd{0.35em}{0.35em}%
\begin{algorithm}[H]
\caption{MitM Detection, Warning, and Warning Persistence}\label{alg:detect}
\KwData{$\mathit{evidence}[]\gets[\,]$; $\mathit{registered}\gets\emptyset$; $\mathit{seen}\gets\emptyset$; $\mathit{absent}[\cdot]\gets 0$; $\mathsf{PeerPin}[\cdot]\gets\bot$ (keyed by peer identity); shares the pinned $pk_{\mathrm{sth}}$ with Algorithm~\ref{alg:client}}
\Fn{\normalfont\textsf{TryPublish}($\mathit{server},\lwarn,\mathit{msg},k_{\mathrm{mac}}$)}{
  \textbf{try} $\textsf{WriteWarning}(\mathit{server},\lwarn,\mathit{msg},k_{\mathrm{mac}})$ \textbf{catch}: report separately and retry next tick \tcp*{the red verdict holds}
}
\Fn{\normalfont\textsf{CheckPeerKeyAndWarn}($\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}},\mathit{expected\_pk}$)}{
  $pk\gets \textsf{RetrieveKey}(\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
  \If{$pk\ne\mathit{expected\_pk}$}{
    \textbf{verdict} $\gets$ \textsc{Alert} \tcp*{preserve the red verdict before publication}
    $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$;\ \ $k_{\mathrm{mac}}\gets \textsf{DeriveWarnMacKey}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
    $\textsf{TryPublish}(\mathit{server},\lwarn,\texttt{"peer-slot-replaced"},k_{\mathrm{mac}})$;\ \ \textbf{raise} \textbf{verdict}\;
  }
  \KwRet $pk$\;
}
\Fn{\normalfont\textsf{RecordWarning}($\mathit{server},\lwarn$)}{
  $(v,\pprefix_w,\pincl_w,\mathit{sth}_w)\gets \mathit{server}.\textsf{Search}(\lwarn)$;\ \ \Req $v\ne\bot$\;
  \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth}_w)$ and the chain $(\pprefix_w,\pincl_w)$ verifies against $\mathit{sth}_w$ \tcp*{evidence is retained only once it verifies}
  $\mathit{evidence}.\textsf{append}((\lwarn,\mathit{sth}_w,\pprefix_w,\pincl_w))$ \tcp*{inclusion proof at the committing epoch}
}
\Fn{\normalfont\textsf{PersistenceCheck}($\mathit{server}$)}{
  \ForEach{$(\lwarn,\mathit{sth}_w,\pprefix_w,\pincl_w)\in \mathit{evidence}$}{
    $(v,\pprefix_{\mathrm{ni}},\pincl_{\mathrm{ni}},\mathit{sth}_{\mathrm{now}})\gets \mathit{server}.\textsf{Search}(\lwarn)$\;
    \If{$v=\bot$}{
      \Req $\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth}_{\mathrm{now}})$ and $(\pprefix_{\mathrm{ni}},\pincl_{\mathrm{ni}})$ proves non-inclusion of $\lwarn$ under $\mathit{sth}_{\mathrm{now}}$ \tcp*{later non-inclusion evidence, a publicly verifiable deletion (\cref{ssec:design-detection})}
      $\pcons\gets \mathit{server}.\textsf{ConsistencyProof}(\mathit{sth}_w.\mathit{tree\_size},\mathit{sth}_{\mathrm{now}}.\mathit{tree\_size})$\;
      \Req $\textsf{VerifyConsistency}(\mathit{sth}_w.\mathit{tree\_size},\mathit{sth}_{\mathrm{now}}.\mathit{tree\_size},\mathit{sth}_w.\mathit{root},\mathit{sth}_{\mathrm{now}}.\mathit{root},\pcons)$\;
      \textbf{raise} DeletionEvidence$(\lwarn,\mathit{sth}_w,\pprefix_w,\pincl_w,\mathit{sth}_{\mathrm{now}},\pprefix_{\mathrm{ni}},\pincl_{\mathrm{ni}},\pcons)$\;
    }
  }
}
\Fn{\normalfont\textsf{BackgroundMonitor}($\mathit{server},\mathit{conversations},\mathit{id_{self}},\mathit{own\_pk}$)}{
  $(v_{\mathsf{vrf}},\mathit{healthy})\gets \textsf{CheckOwnVrfKey}()$ \tcp*{global VRF identity self-check, or ends the tick if unreachable}
  \If{$\neg\,\mathit{healthy}$}{
    \textbf{verdict} $\gets$ \textsc{Alert} \tcp*{preserve the red verdict before publication}
    \ForEach{$(\sharedS,\mathit{id_{peer}})\in \mathit{conversations}$}{
      $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$;\ \ $k_{\mathrm{mac}}\gets \textsf{DeriveWarnMacKey}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$\;
      $\textsf{TryPublish}(\mathit{server},\lwarn,\texttt{"key-replaced"},k_{\mathrm{mac}})$\;
    }
    \textbf{emit} \textbf{verdict} \tcp*{red emitted, then the sweep runs}
  }
  \ForEach{$(\sharedS,\mathit{id_{peer}})\in \mathit{conversations}$}{
    $(\lself,\lpeer,\lwarn)\gets \textsf{DeriveLabels}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$;\ \ $k_{\mathrm{mac}}\gets \textsf{DeriveWarnMacKey}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$ \tcp*{secret-bound labels and warn-MAC key}
    \lIf{$\lself\notin \mathit{registered}$}{$\textsf{Register}(\mathit{server},\sharedS,\mathit{id_{self}},\mathit{id_{peer}},\mathit{own\_pk})$;\ \ $\mathit{registered}\gets \mathit{registered}\cup\{\lself\}$ \tcp*[f]{once per derived label}}
    $\textsf{WindowedCatchUp}(\mathit{server})$ \tcp*{append-only and STH growth}
    $v\gets \textsf{VerifiedLookup}(\mathit{server},\lself)$ \tcp*{owner slot}
    \lIf{$v\ne\mathit{own\_pk}$}{\textbf{verdict} $\gets$ \textsc{Alert};\ \ $\textsf{TryPublish}(\mathit{server},\lwarn,\texttt{"lookup-replaced"},k_{\mathrm{mac}})$;\ \ \textbf{emit} \textbf{verdict};\ \ \textbf{continue}}
    \lIf{$\textsf{CheckWarning}(\mathit{server},\lwarn,k_{\mathrm{mac}})=\textsc{Alert}$}{\textbf{emit} \textsc{Alert};\ \ \textbf{continue} \tcp*[f]{warning slot, checked before peer absence}}
    $pk\gets \textsf{VerifiedLookup}(\mathit{server},\lpeer)$ \tcp*{peer slot ($pk=\bot$ is proven non-inclusion)}
    \uIf{$pk=\bot$}{
      \uIf{$\lpeer\in \mathit{seen}$}{\textbf{emit} \textsc{Alert};\ \ \textbf{continue} \tcp*{removal after prior presence}}
      \uElseIf{$\mathit{absent}[\lpeer]<2$}{$\mathit{absent}[\lpeer]\gets \mathit{absent}[\lpeer]+1$;\ \ \textbf{emit} pending state;\ \ \textbf{continue} \tcp*{checks 1--2: pending, no UI event}}
      \Else{\textbf{emit} operational \texttt{NotPosted} for this conversation;\ \ \textbf{continue} \tcp*{third check: fail-closed timeout}}
    }\Else{
      $\mathit{absent}[\lpeer]\gets 0$;\ \ $\mathit{seen}\gets \mathit{seen}\cup\{\lpeer\}$ \tcp*{present: reset the absence counter}
      $\mathit{pinned}\gets\mathsf{PeerPin}[\mathit{id_{peer}}]$\;
      \uIf{$\mathit{pinned}=\bot$}{$\mathsf{PeerPin}[\mathit{id_{peer}}]\gets pk$ \tcp*[f]{first sight: pin the peer key}}
      \uElseIf{$pk\ne\mathit{pinned}$}{\textbf{verdict} $\gets$ \textsc{Alert};\ \ $\textsf{TryPublish}(\mathit{server},\lwarn,\texttt{"peer-slot-replaced"},k_{\mathrm{mac}})$;\ \ \textbf{emit} \textbf{verdict};\ \ \textbf{continue}}
    }
    \lIf{$\mathit{healthy}$}{\textbf{emit} a green verdict \tcp*[f]{green only when the self-check is healthy}}
  }
}
\end{algorithm}

\paragraph{The auditor (Algorithm~\ref{alg:auditor}).} The auditor is an external verifier that replays
each declared transition on a temporary copy of its accepted directory state. DUET begins this replay
at epoch~0 and advances the accepted state, cursor, and signed-head baseline together only after every
stage succeeds. A failed delta fetch leaves them unchanged so that the epoch can be retried. This
update flow follows the IETF KeyTrans \texttt{AuditorUpdate} pattern, in which the service operator
supplies an update for each new log entry to a third-party auditor~\cite[Section~15.2]{ietf-keytrans-protocol}.

Split-view detection requires comparison with a signed head obtained from an independent vantage.
The A4 demonstration instead uses a staged second signed head supplied by the same directory server.
Therefore, it tests the signed-head comparison only. Two validly signed heads at the same tree size
with different roots produce \textsc{Fork}. Heads of different sizes require a valid consistency proof.
A missing or invalid proof produces \textsc{ConsistencyUnverified} without establishing equivocation.
If no valid comparison head is available, a successful transition audit returns
\textsc{ConsistentNoVantage}.

{\SetAlFnt{\footnotesize}\SetInd{0.35em}{0.35em}%
\begin{algorithm}[H]
\caption{Networked auditor: per-epoch audit (\textsf{AuditEpoch})}\label{alg:auditor}
\KwData{last trusted signed head $\mathit{last\_sth}$ (initially $\bot$); committed $\mathit{replica}$ (a PrefixTree,
empty initially); audited-epoch cursor $\mathit{audited\_epochs}\gets0$; trusted STH key $pk_{\mathrm{sth}}$; expected configuration commitment $\mathit{expected\_config\_hash}$}
\Fn{\normalfont\textsf{AuditEpoch}($\mathit{server}$)}{
  $\mathit{sth_{new}}\gets \mathit{server}.\textsf{GetCurrentSTH}()$\;
  \lIf{$\mathit{sth_{new}}=\bot$}{\KwRet \textsc{Unavailable} \tcp*[f]{retryable}}
  \lIf{$\neg\,\STH.\textsf{Verify}(pk_{\mathrm{sth}},\mathit{sth_{new}})\ \vee\ \mathit{sth_{new}}.\mathit{config\_hash}\ne\mathit{expected\_config\_hash}\ \vee\ \mathit{sth_{new}}.\mathit{epoch}+1\ne \mathit{sth_{new}}.\mathit{tree\_size}$}{\textbf{alarm};\ \KwRet \textsc{Malformed}}
  \If{$\mathit{last\_sth}\ne\bot$}{
    \lIf{$\mathit{sth_{new}}.\mathit{epoch}<\mathit{last\_sth}.\mathit{epoch}\ \vee\ \mathit{sth_{new}}.\mathit{tree\_size}<\mathit{last\_sth}.\mathit{tree\_size}$}{\textbf{alarm};\ \KwRet \textsc{Rollback}}
    \lIf{$\mathit{sth_{new}}.\mathit{tree\_size}=\mathit{last\_sth}.\mathit{tree\_size}\ \wedge\ \mathit{sth_{new}}.\mathit{root}\ne\mathit{last\_sth}.\mathit{root}$}{\textbf{alarm};\ \KwRet \textsc{Fork}}
    \If{$\mathit{sth_{new}}.\mathit{tree\_size}>\mathit{last\_sth}.\mathit{tree\_size}$}{
      $\pcons\gets \mathit{server}.\textsf{ConsistencyProof}(\mathit{last\_sth}.\mathit{tree\_size},\mathit{sth_{new}}.\mathit{tree\_size})$\;
      \lIf{$\pcons=\bot\ \vee\ \neg\,\textsf{VerifyConsistency}(\mathit{last\_sth}.\mathit{tree\_size},\mathit{sth_{new}}.\mathit{tree\_size},\mathit{last\_sth}.\mathit{root},\mathit{sth_{new}}.\mathit{root},\pcons)$}{\textbf{alarm};\ \KwRet \textsc{ConsistencyUnverified} \tcp*[f]{append-only unverified}}
    }
  }
  $\mathit{candidate}\gets \textsf{clone}(\mathit{replica})$;\ \ $\mathit{cursor}\gets \mathit{audited\_epochs}$;\ \ $\mathit{complete}\gets \textbf{true}$\;
  \ForEach{not-yet-audited epoch $e\le \mathit{sth_{new}}.\mathit{epoch}$ \emph{(including $e=0$)}}{
    $\mathit{delta}_e\gets \mathit{server}.\textsf{GetEpochDelta}(e,\mathit{sth_{new}}.\mathit{tree\_size})$\;
    \lIf{$\mathit{delta}_e = \bot$}{$\mathit{complete}\gets \textbf{false}$;\ \textbf{break} \tcp*[f]{fetch failure: retryable}}
    \lIf{$\textsf{AuditTransition}(\mathit{candidate},e,\mathit{delta}_e,\mathit{sth_{new}}.\mathit{root},\mathit{sth_{new}}.\mathit{tree\_size})=\textsc{TransitionFailed}$}{\textbf{alarm};\ \KwRet \textsc{TransitionFailed}}
    $\mathit{cursor}\gets \mathit{cursor}+1$\;
  }
  \lIf{$\neg\,\mathit{complete}$}{\KwRet \textsc{Incomplete}}
  $\mathit{vb}\gets\textsf{DetectFork}(\mathit{server}.\textsf{GetComparisonHead}(),\mathit{sth_{new}},\mathit{server})$ \tcp*{comparison check only after complete replay}
  \lIf{$\mathit{vb}=\textsc{Fork}$}{\textbf{alarm};\ \KwRet \textsc{Fork}}
  \lIf{$\mathit{vb}=\textsc{ConsistencyUnverified}$}{\textbf{alarm};\ \KwRet \textsc{ConsistencyUnverified} \tcp*[f]{hold, no fork}}
  $\mathit{replica}\gets \mathit{candidate}$;\ \ $\mathit{audited\_epochs}\gets \mathit{cursor}$;\ \ $\mathit{last\_sth}\gets \mathit{sth_{new}}$ \tcp*{advance the copy, cursor, and baseline together}
  \lIf{$\mathit{vb}=\textsc{InvalidHead}$}{\KwRet \textsc{ConsistentNoVantage} \tcp*[f]{committed, no split-view verdict}}
  \KwRet \textsc{Consistent}\;
}
\end{algorithm}

\paragraph{Continuous mode (Algorithm~\ref{alg:cea}).} CEA authenticates the Double Ratchet epoch by
epoch, following ACKA's CEA construction and composition order~\cite[Figures~8 and~10]{dowling2023acka}.
It publishes a directional fingerprint over the current public ratchet key $T_t$, which the receiver
recomputes and compares, and folds the epoch's secret input $I_t$ into the chained key $k_t$ only after
the directory operation succeeds. A mismatch shows that the committed fingerprint is inconsistent with
the receiver's chained state and received ratchet public key $T_t$. The verifier reports this as
\textsc{Fork} without attributing the divergence to a particular input.

Since the receiver does not repeat the lookup after accepting the epoch, the transition audit detects
a later operator-side overwrite of the immutable fingerprint. The label uses the canonical $\min/\max$ identity order, so both
parties address the same slot in an epoch. The fingerprint remains directional (sender-then-receiver)
and is computed over the current, pre-update state.

Both sender and receiver read the epoch slot through \textsf{VerifiedCeaLookup}, which runs the full
proof chain of \textsf{VerifiedLookup} (Algorithm~\ref{alg:client}, without the VRF step) against the
signed head returned with the lookup. If no baseline exists, the verified head establishes the
trust-on-first-use baseline. Otherwise, a read-only, bidirectional continuity check relates the
returned head to the retained history. CEA reads can interleave with directory commits and monitor
checks, so the returned head may precede or follow the retained baseline. The check proves continuity
in the applicable direction without advancing that baseline.

The receiver retries a proof-verified absence before returning \textsc{NotPosted}. Before writing, the
sender verifies that its label is unoccupied. An occupied label indicates impersonation. The sender
advances its chain only after a verified read-back confirms the committed fingerprint. The receiver
advances only on a \textsc{Consistent} verdict and leaves its state unchanged on \textsc{Fork} or
\textsc{NotPosted}.

\begin{algorithm}[H]
\caption{ACKA Continuous Entity Authentication Mode}\label{alg:cea}
\KwData{per conversation: shared secret $\sharedS$ and identities $\mathit{id_{self}},\mathit{id_{peer}}$; chained secret $k_t$; epoch $t$; warning label $\lwarn$ and warn-MAC key $k_{\mathrm{mac}}\gets\textsf{DeriveWarnMacKey}(\sharedS,\mathit{id_{self}},\mathit{id_{peer}})$ (Alg.~\ref{alg:dualindex})}
\Fn{\normalfont\textsf{VerifiedCeaLookup}($\mathit{server},L$)}{
  $(v,\pprefix,\pincl,\mathit{sth})\gets \mathit{server}.\textsf{Search}(L)$\;
  run the raw-label checks of \textsf{VerifiedLookup} (Alg.~\ref{alg:client}, no VRF): signed head, expected configuration, label and value binding, immutable kind on a present leaf, inclusion at the current size\;
  \uIf{no baseline yet}{adopt $\mathit{sth}$ as the trust-on-first-use baseline}
  \lElse{\Req $\mathit{sth}$ chains from the baseline in whichever direction holds, without advancing it}
  \KwRet $v$ \tcp*{$v=\bot$ only for verified non-inclusion, an invalid proof or head is rejected}
}
\Fn{\normalfont\textsf{CeaSetup}($\mathit{pss},\mathit{id}_{\A},\mathit{id}_{\B}$)}{
  $k_0\gets \KDF(\mathit{pss},\min(\mathit{id}_{\A},\mathit{id}_{\B})\concat \max(\mathit{id}_{\A},\mathit{id}_{\B}))$;\ \ $t\gets 0$\;
}
\Fn{\normalfont\textsf{CeaUpdate}($I_t$)}{
  $k_{t+1}\gets \KDF(k_t,\KDF(I_t,\texttt{"CEA"}))$;\ \ $t\gets t+1$\;
}
\Fn{\normalfont\textsf{CeaLprint}($\mathit{id}_{\A},\mathit{id}_{\B}$)}{
  \KwRet $\Hash(k_t\concat \min(\mathit{id}_{\A},\mathit{id}_{\B})\concat \max(\mathit{id}_{\A},\mathit{id}_{\B})\concat t)$\;
}
\Fn{\normalfont\textsf{CeaFprint}($\mathit{id_{sender}},\mathit{id_{peer}},T_t$)}{
  $\mathit{fkey}\gets \KDF(k_t,\mathit{id_{sender}}\concat \mathit{id_{peer}}\concat t)$\;
  \KwRet $\Hash(\mathit{fkey}\concat T_t)$\;
}
\Fn{\normalfont\textsf{CeaRegister}($\mathit{server},\mathit{id_{self}},\mathit{id_{peer}},T_t,I_t$)}{
  $L_t\gets \textsf{CeaLprint}(\mathit{id_{self}},\mathit{id_{peer}})$;\ \ $F_t\gets \textsf{CeaFprint}(\mathit{id_{self}},\mathit{id_{peer}},T_t)$\;
  \If{$\textsf{VerifiedCeaLookup}(\mathit{server},L_t)\ne\bot$}{
    \textbf{raise} \textsc{Alert}\;
  }
  $\mathit{server}.\textsf{UpdateRaw}(L_t,F_t)$;\ \ $\mathit{server}.\textsf{CommitEpoch}()$\;
  \Req $\textsf{VerifiedCeaLookup}(\mathit{server},L_t)=F_t$ \tcp*{confirm under the head returned with this read}
  $\textsf{CeaUpdate}(I_t)$ \tcp*{advance after the confirmed write}
}
\Fn{\normalfont\textsf{CeaVerify}($\mathit{server},\mathit{id_{sender}},\mathit{id_{self}},T_t,I_t,\lwarn$)}{
  $L_t\gets \textsf{CeaLprint}(\mathit{id_{sender}},\mathit{id_{self}})$;\ \ $F^{*}_t\gets \textsf{CeaFprint}(\mathit{id_{sender}},\mathit{id_{self}},T_t)$\;
  $v\gets \textsf{VerifiedCeaLookup}(\mathit{server},L_t)$ \tcp*{chain and read-only bidirectional anchor, no VRF, retried on \textsc{NotPosted} before concluding}
  \lIf{$v=\bot$}{\KwRet \textsc{NotPosted} \tcp*[f]{no advance}}
  \If{$v\ne F^{*}_t$}{
    $\mathit{verdict}\gets\textsc{Fork}$\;
    \textbf{try} $\textsf{WriteWarning}(\mathit{server},\lwarn,\texttt{"ratchet-fork"},k_{\mathrm{mac}})$ \textbf{catch}: report the failure separately\;
    \KwRet $\mathit{verdict}$ \tcp*{no advance, verdict independent of publication}
  }
  $\textsf{CeaUpdate}(I_t)$ \tcp*{advance on \textsc{Consistent}}
  \KwRet \textsc{Consistent}\;
}
\end{algorithm}
