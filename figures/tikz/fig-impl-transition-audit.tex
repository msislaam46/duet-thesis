% fig-impl-transition-audit.tex — the transition auditor's state flow (Alg 7).
% Covers the first-head bootstrap, the append-only consistency check for a grown
% head, the per-epoch replay loop, and the two equal-head cases (already audited
% vs. an incomplete transition that must be retried). Label: \label{fig:impl-transition-audit}
\begin{tikzpicture}[>=Stealth,font=\scriptsize,node distance=5.0mm,
  proc/.style={draw,rounded corners,align=center,inner sep=3pt,minimum width=6.0cm,minimum height=7mm},
  dec/.style={draw,align=center,inner sep=3pt,minimum width=6.0cm,minimum height=7mm,fill=blue!6},
  bad/.style={draw,rounded corners,align=center,inner sep=3pt,draw=red!60!black,fill=red!3,text width=3.4cm},
  rtr/.style={draw,rounded corners,align=center,inner sep=3pt,draw=orange!80!black,fill=orange!4,text width=3.4cm},
  good/.style={draw,rounded corners,align=center,inner sep=3pt,draw=green!50!black,fill=green!4,minimum width=6.0cm}]

  \node[proc] (n1) {receive new head; verify STH signature};
  \node[dec,below=of n1] (nf) {first head? (no recorded baseline)};
  \node[dec,below=of nf] (n2) {rollback, or equal size with a different root?};
  \node[dec,below=of n2] (nc) {log grew (new size $>$ baseline)?};
  \node[proc,below=of nc] (ncp) {fetch $+$ verify RFC~9162 consistency proof\\(append-only, baseline\,$\to$\,new head)};
  \node[proc,below=of ncp,minimum height=13mm] (loop) {\textbf{for each not-yet-audited epoch} $e$ (cursor\,$\to$\,new, incl.\ $e{=}0$):\\clone mirror into scratch; fetch delta$_e$; replay each Insert / Rotate\\(A1: reject duplicate insert; C2: reject immutable rotation);\\require replayed scratch root $=$ declared root$_e$};
  \node[good,below=of loop] (n7) {every epoch replayed: commit mirror, cursor, head atomically};

  \draw[->] (n1) -- (nf);
  \draw[->] (nf) -- node[left,font=\tiny]{no} (n2);
  \draw[->] (n2) -- node[left,font=\tiny]{no} (nc);
  \draw[->] (nc) -- node[left,font=\tiny]{yes} (ncp);
  \draw[->] (ncp) -- (loop);
  \draw[->] (loop) -- (n7);

  % ---- side branches (right) ----
  \node[good,right=9mm of nf] (bf) {yes: trust-on-first-use, adopt as baseline (no consistency or transition check)};
  \draw[->] (nf.east) -- node[above,font=\tiny]{yes} (bf.west);
  \node[bad,right=9mm of n2] (b2) {yes: reject, raise Fork};
  \draw[->] (n2.east) -- node[above,font=\tiny]{yes} (b2.west);
  \node[dec,right=9mm of nc,minimum width=3.4cm,align=center] (nq) {equal head: all epochs already audited?};
  \draw[->] (nc.east) -- node[above,font=\tiny]{no} (nq.west);
  \node[rtr,above=5mm of nq] (bnc) {yes: idempotent re-poll, no state change};
  \draw[->] (nq.north) -- node[right,font=\tiny]{yes} (bnc.south);
  \draw[->] (nq.south) |- node[pos=0.25,right,font=\tiny]{no: retry unaudited transition} (loop.east);
  \node[bad,right=9mm of ncp] (bcp) {fail: ALARM (latch), no replay};
  \draw[->] (ncp.east) -- node[above,font=\tiny]{invalid} (bcp.west);
  \node[bad,below=8mm of bcp] (b6) {root mismatch / A1 / C2: ALARM, transition audit failed};
  \draw[->] (loop.east) -- node[above,font=\tiny]{no} (b6.west);
  \node[rtr,below=5mm of b6] (b3) {delta-fetch failure: discard scratch, retry (incomplete, not a fork)};
  \draw[->] (loop.south east) -- node[below,font=\tiny,pos=0.2]{fetch fail} (b3.west);
\end{tikzpicture}
