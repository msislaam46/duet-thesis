% fig-impl-transition-audit.tex — the transition auditor's state flow (Alg 7),
% split into two coordinated diagrams: (a) head classification and (b) the
% transactional per-epoch transition replay. The first head skips only the
% prior-head consistency proof; it still replays epochs 0..current before the
% baseline is adopted. Label: \label{fig:impl-transition-audit}
\begin{tikzpicture}[>=Stealth,font=\scriptsize,node distance=4.5mm,
  proc/.style={draw,rounded corners,align=center,inner sep=3pt,text width=4.7cm,minimum height=6mm},
  dec/.style={draw,align=center,inner sep=3pt,text width=4.7cm,minimum height=6mm,fill=blue!6},
  term/.style={draw,rounded corners,align=center,inner sep=3pt,text width=3.6cm,minimum height=6mm},
  bad/.style={term,draw=red!60!black,fill=red!3},
  nop/.style={term,draw=orange!80!black,fill=orange!4},
  go/.style={term,draw=green!50!black,fill=green!5}]

  % ---- (a) head classification ----
  \node[proc] (a0) {\textbf{(a) head classification}\\receive new head; verify STH signature};
  \node[dec,below=of a0] (a1) {first head? (no recorded baseline)};
  \node[dec,below=of a1] (a2) {rollback, or equal size with a different root?};
  \node[dec,below=of a2] (a3) {log grew (new size $>$ baseline)?};
  \node[proc,below=of a3] (a3p) {verify RFC~9162 consistency proof (baseline\,$\to$\,head)};
  \node[dec,below=of a3p] (a4) {equal head: all epochs already audited?};
  \draw[->] (a0) -- (a1);
  \draw[->] (a1) -- node[left,font=\tiny]{no} (a2);
  \draw[->] (a2) -- node[left,font=\tiny]{no} (a3);
  \draw[->] (a3) -- node[left,font=\tiny]{yes} (a3p);
  \draw[->] (a3p) -- (a4);
  % terminals on the right, well clear of the spine
  \node[go,right=10mm of a1] (t1) {yes: adopt baseline, skip prior-head consistency proof; still replay $\to$ (b)};
  \draw[->] (a1.east) -- node[above,font=\tiny]{yes} (t1.west);
  \node[bad,right=10mm of a2] (t2) {yes: ALARM: Fork};
  \draw[->] (a2.east) -- node[above,font=\tiny]{yes} (t2.west);
  \node[bad,right=10mm of a3p] (t3) {invalid proof: ALARM (latch), no replay};
  \draw[->] (a3p.east) -- node[above,font=\tiny]{fail} (t3.west);
  \node[nop,above right=1mm and 10mm of a4] (t4) {yes: idempotent re-poll, no state change};
  \draw[->] (a4.east) -- node[above,font=\tiny,pos=0.3]{yes} (t4.west);
  \node[go,below=of a4] (t5) {no: incomplete $\to$ replay (b)};
  \draw[->] (a4) -- node[left,font=\tiny]{no} (t5);
\end{tikzpicture}

  % ---- (b) transactional transition replay ----
  \node[proc] (b0) {\textbf{(b) transactional transition replay} (entered from (a))\\clone the committed mirror into scratch; $\mathit{cursor}\gets\mathit{audited\_epochs}$};
  \node[loop,below=of b0] (b1) {\textbf{for each not-yet-audited epoch} $e$ (cursor\,$\to$\,current, incl.\ $e{=}0$):\\fetch delta$_e$; replay each Insert / Rotate (reject a duplicate insert (A1) or an immutable rotation (C2)); require scratch root $=$ declared root$_e$};
  \node[good,below=of b1] (b2) {all epochs replayed: commit mirror, cursor, head atomically};
  \draw[->] (b0) -- (b1);
  \draw[->] (b1) -- (b2);
  \node[rtr,right=10mm of b0] (r1) {delta-fetch failure: discard scratch, retry (incomplete, not a fork)};
  \draw[->] (b1.north east) -- node[above,font=\tiny,pos=0.4]{fetch fail} (r1.west);
  \node[bad,right=10mm of b1] (r2) {root mismatch / A1 / C2: ALARM, transition audit failed};
  \draw[->] (b1.east) -- node[above,font=\tiny]{fail} (r2.west);
\end{tikzpicture}
