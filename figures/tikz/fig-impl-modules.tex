% fig-impl-modules.tex — Rust architecture by ownership.
% The server core (KTServer) composes the prefix tree, Merkle log, ECVRF, and the
% Ed25519 signer; only the prefix tree and Merkle log hash through the project's
% SHA-256 component. Client-side constructions (dual-index derivation feeding the
% CEA registration/verification) call the server API; the auditor is a separate
% process that keeps its own replica.
% Label: \label{fig:impl-modules}
\resizebox{\textwidth}{!}{%
\begin{tikzpicture}[>=Stealth,font=\scriptsize,
   cmp/.style={draw,rounded corners,align=center,minimum height=8mm,inner sep=3pt,fill=white},
   grp/.style={draw,dashed,rounded corners,inner sep=6pt},
   band/.style={font={\footnotesize\itshape},anchor=east,text=black!70}]

  % ---- server stack (left/centre) ----
  \node[cmp,minimum width=5.0cm] (api) at (0,5.3) {\textbf{HTTP server}\\lookup, update, and consistency API};
  \node[cmp,minimum width=3.0cm] (dir) at (0,3.8) {\textbf{Directory service}\\(\texttt{KTServer})};
  % KTServer's real dependencies
  \node[cmp,minimum width=2.3cm] (ptree) at (-3.6,2.1) {Sparse Merkle\\prefix tree};
  \node[cmp,minimum width=2.1cm] (mlog)  at (-1.2,2.1) {Append-only\\Merkle log};
  \node[cmp,minimum width=1.9cm] (vrf)   at ( 1.0,2.1) {ECVRF};
  \node[cmp,minimum width=2.3cm] (sig)   at ( 3.3,2.1) {Ed25519\\signer};
  \node[cmp,minimum width=8.0cm] (hash)  at (-0.15,0.5) {SHA-256 hashing};
  % bands
  \node[band] at (-5.2,5.3) {API};
  \node[band] at (-5.2,3.8) {server core};
  \node[band,align=right] at (-5.2,2.1) {authenticated\\structures};
  \node[band] at (-5.2,0.5) {hashing layer};
  % server dependency edges
  \draw[->] (api) -- (dir);
  \foreach \n in {ptree,mlog,vrf,sig}{\draw[->] (dir) -- (\n);}
  \foreach \n in {ptree,mlog}{\draw[->] (\n) -- (\n |- hash.north);}

  % ---- client-side constructions (right) : they CALL the API ----
  \node[cmp,minimum width=3.0cm] (didx) at (6.6,4.2) {Dual-index\\derivation};
  \node[cmp,minimum width=3.0cm] (cea)  at (6.6,2.9) {CEA registration\\and verification};
  \node[grp,fit=(didx)(cea),label={[font=\footnotesize\itshape,text=black!70]above:Client-side constructions}] (cg) {};
  \draw[->,dashed] (didx) -- (cea);
  \draw[->,dashed] (cea.west) -- node[above,font=\scriptsize,pos=0.55]{call the API} (api.east);

  % ---- auditor process ----
  \node[cmp,minimum width=3.0cm,fill=black!5] (aud) at (7.6,6.0) {\textbf{Auditor replica}\\separate process};
  \draw[<->,dashed] (aud.west) -- node[above,align=center,font=\scriptsize,pos=0.55]{obtain signed heads,\\epoch deltas, and\\consistency proofs} (api.north east);
\end{tikzpicture}%
}
