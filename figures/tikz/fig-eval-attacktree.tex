% fig-eval-attacktree.tex — AND/OR attack tree (reading Alice--Bob plaintext).
% Reading plaintext requires BOTH interposing a key AND evading detection; the
% evasion actions (A2 censorship, A4 equivocation) are not plaintext-reveal paths
% on their own. Label: \label{fig:eval-attacktree}
\begin{tikzpicture}[>=Stealth,font=\scriptsize,
  goal/.style={draw,thick,rounded corners,fill=black!6,align=center,minimum height=8mm,inner sep=4pt},
  gate/.style={draw,rounded corners,fill=blue!8,draw=blue!55!black,align=center,inner sep=3pt,font=\scriptsize\bfseries},
  atk/.style={draw,rounded corners,draw=red!60!black,align=center,minimum height=8mm,minimum width=2.3cm,inner sep=3pt},
  det/.style={font=\scriptsize,text=green!45!black,align=center},
  oos/.style={draw,dashed,rounded corners,draw=orange!80!black,align=center,minimum height=7mm,inner sep=4pt}]

  % goal and top AND gate
  \node[goal] (g)    at (0,4.4) {Goal: read Alice--Bob plaintext};
  \node[gate] (and0) at (0,3.2) {AND \textnormal{(both required)}};
  \draw[->,black!45] (g) -- (and0);

  % two sub-goals
  \node[gate] (orA)  at (-4.2,2.0) {OR\textnormal{: interpose a key}};
  \node[gate] (andB) at ( 4.2,2.0) {AND\textnormal{: evade detection}};
  \draw[->,black!45] (and0) -- (orA);
  \draw[->,black!45] (and0) -- (andB);

  % interposition leaves (OR)
  \node[atk] (a1) at (-6.6,0.4) {A1 server\\substitution\\{\tiny T1565.002}};
  \node[atk] (a3) at (-4.2,0.4) {A3 network\\MitM\\{\tiny T1557}};
  \node[atk] (a5) at (-1.8,0.4) {A5 malicious\\rotation\\{\tiny T1565.002}};
  \foreach \n in {a1,a3,a5}{\draw[->,black!45] (orA) -- (\n);}
  \node[det] at (-4.2,-1.2) {each detected on its own: self-check,\\label mismatch, or identity re-fetch $\to$ warning};

  % evasion leaves (AND)
  \node[atk] (a2) at (2.6,0.4) {A2 suppress\\warning};
  \node[atk] (a4) at (5.8,0.4) {A4 equivocate /\\hide from auditor};
  \draw[->,black!45] (andB) -- (a2);
  \draw[->,black!45] (andB) -- (a4);
  \node[det] at (4.2,-1.2) {each detected: compound suppression\\signal (A2), consistency + auditor (A4)};

  % out of scope
  \node[oos] (x1) at (0,-2.6) {X1 endpoint compromise: outside the KT trust boundary, not detectable (T1552.004)};
\end{tikzpicture}
